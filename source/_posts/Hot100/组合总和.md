# Code

```c++
class Solution {
public:
    vector<vector<int>>ans;
    vector<int>t;
    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        sort(candidates.begin(),candidates.end());
        dfs(candidates,target,0);
        return ans;
    }
    void dfs(vector<int>& candidates, int target,int u)
    {
        if(target==0)
        {
            ans.push_back(t);
            return;
        }
        if(u>=candidates.size())return;//保证下面代码的u合法
        if(target<candidates[u])return;//排序之后的剪枝
        
        t.push_back(candidates[u]);
        //dfs只有两种情况，选当前这个或者不选（由于每个数可以选择多次，因此选择时这里k不能加1）
        dfs(candidates,target-candidates[u],u);
        t.pop_back();
        dfs(candidates,target,u+1);
        
    }
};
```
