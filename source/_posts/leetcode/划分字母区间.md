# Code

```c++
class Solution {
public:
    vector<int> partitionLabels(string s) {
        unordered_map<char,int>h;
        for(int i=0;i<s.size();i++)
            h[s[i]]=i;
        
        vector<int>ans;
        int st=0,ed=0;
        for(int i=0;i<s.size();i++)
        {
            ed=max(ed,h[s[i]]);
            if(i==ed)
            {
                ans.push_back(i-st+1);
                st=i+1;
            }
        }
        return ans;
    }
};
```
