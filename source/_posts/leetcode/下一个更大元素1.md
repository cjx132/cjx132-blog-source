# 综述

-   单调栈模板题：求右边第一个xxx：倒序，左边第一个xxx：正序
-   先在nums\[2\]中跑一遍单调栈，并记录结果到哈希表，之后遍历nums1，求结果即可

# Code

```c++
class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        int n2=nums2.size();
        stack<int>st;
        unordered_map<int,int>h;
        for(int i=n2-1;i>=0;i--)
        {
            while(st.size()&&nums2[i]>nums2[st.top()])st.pop();
            if(st.size())h[nums2[i]]=st.top();
            else h[nums2[i]]=-1;
            st.push(i);
        }
        vector<int>ans;
        int n1=nums1.size();
        for(int i=0;i<n1;i++)
        {
            int x=nums1[i];
            if(h.count(x)&&h[x]!=-1)ans.push_back(nums2[h[x]]);
            else ans.push_back(-1);
        }
        return ans;
    }
};
```
