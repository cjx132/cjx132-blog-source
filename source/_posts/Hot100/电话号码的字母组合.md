# Code

```c++
class Solution {
public:
    vector<string>res;
    string t;
    unordered_map<char, string> mp{
        {'2', "abc"},{'3',"def"}, {'4',"ghi"}, 
        {'5',"jkl"},{'6',"mno"}, {'7',"pqrs"},
        {'8',"tuv"},{'9',"wxyz"}}; 
    vector<string> letterCombinations(string digits) {
        if(digits.size()==0)return res;
        dfs(digits,0);
        return res;
    }
    void dfs(string &digits,int u)
    {
        if(u==digits.size())
        {
            res.push_back(t);
            return;
        }
        string str=mp[digits[u]];//在每一个位置处，可能的字符只可能为digit[k]所代表的几个字母
        for(char c:str)
        {
            t+=c;
            dfs(digits,u+1);
            t.pop_back();
        }
    }
};
```
