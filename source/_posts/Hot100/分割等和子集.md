dp \[i\] \[j\] 表示从数组的 \[0, i\] 这个子区间内挑选一些正整数，每个数只能用一次，使得这些数的和**恰好等于** j

# Code

```c++
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int len=nums.size();
        if(len==0)return false;
        int sum=0;
        for(int num :nums){
            sum+=num;
        }
        if(sum%2)return false;
        int target=sum/2;
        //恰好等于所以初始化为false
        vector<vector<int> >dp(len,vector<int>(target+1,0));
        //第0行只有dp[0][nums[0]]才为true
        if(nums[0]<=target){
            dp[0][nums[0]]=1;
        }
        for(int i=1;i<len;i++){
            //二维dp时正序或倒序都可，空间优化为一维dp时只能倒序
            for(int j=target;j>=0;j--){
                //先照抄下来
                dp[i][j]=dp[i-1][j];
                if(nums[i]<=j){
                    dp[i][j]=dp[i-1][j]||dp[i-1][j-nums[i]];
                }
            }
            //剪枝，提前满足条件
            if(dp[i][target]==1)return 1;
        }
        return dp[len-1][target];
    }
};
```
```c++
//优化空间复杂度代码
//需要熟记下面代码，为0-1背包的经典解法
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int len=nums.size();
        if(len==0)return false;
        int sum=0;
        for(int num :nums){
            sum+=num;
        }
        if(sum%2)return false;
        int target=sum/2;
        //需要注意恰好和小于的初始化问题
        vector<int>dp(target+1,0);
        if(nums[0]<=target){
            dp[nums[0]]=1;
        }
        //0-1背包的核心解法
        for(int i=1;i<len;i++){
            /*这段代码可以看出nums[i]>j时dp[j]新值与老值相等，因此可简化!
            for(int j=target;j>=0;j--){
                dp[j]=dp[j];
                if(nums[i]<=j){//nums[i]>j时不进入此分支
                    dp[j]=dp[j]||dp[j-nums[i]];
                }
            }*/
            
            //此段优化是倒序遍历所特有的
            for(int j=target;j>=nums[i];j--){
                dp[j]=dp[j]||dp[j-nums[i]];
            }
            
            if(dp[target]==1)return 1;
        }
        return dp[target];
    }
};
```

# 复杂度分析

-   时间复杂度：O(NC)，这里 N 是数组元素的个数，C是数组元素的和的一半
    
-   空间复杂度：O(NC)，优化后O(C)
