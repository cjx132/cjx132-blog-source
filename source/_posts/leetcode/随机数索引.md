# Code

```c++
class Solution {
public:
    vector<int>arr;
    Solution(vector<int>& nums) {
        arr=nums;
    }
    
    int pick(int target) {
        //蓄水池容量为1
        int res;
        int cnt=0;
        for(int i=0;i<arr.size();i++){
            if(arr[i]==target)cnt++;//cnt计数
            if(cnt==1&&arr[i]==target)res=i;//第一个直接放入蓄水池
            else if(cnt>1&&arr[i]==target){
                int d=rand()%cnt;
                if(d==0)res=i;
            }
        }
        return res;
    }
};

/**
 * Your Solution object will be instantiated and called as such:
 * Solution* obj = new Solution(nums);
 * int param_1 = obj->pick(target);
 */
```
