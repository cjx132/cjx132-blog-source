# 综述

-   类似于快排中的划分，但是while中的i<j不能少

# Code

```c++
class Solution {
public:
    vector<int> exchange(vector<int>& nums) {
        int i=0,j=nums.size()-1;
        while(i<j)
        {
            while(i<j&&nums[j]%2==0)j--;
            while(i<j&&nums[i]%2==1)i++;
            if(i<j)swap(nums[i],nums[j]);
        }
        return nums;
    }
};
```
