```c++
class Solution {
public:
    vector<vector<int>> updateMatrix(vector<vector<int>>& matrix) {
        int m=matrix.size();
        int n=matrix[0].size();
        vector<vector<int>>ans=vector<vector<int>>(m,vector<int>(n,0));
        vector<vector<int>>visit=vector<vector<int>>(m,vector<int>(n,0));
        int dx[4]={1,-1,0,0},dy[4]={0,0,-1,1};
        queue<pair<int,int>> q;
        for(int i=0;i<m;i++){
            for(int j=0;j<n;j++){
                if(matrix[i][j]==0){
                    q.push({i,j});
                    visit[i][j] = 1;
                }
            }
        }   
        int cntd = 0;
        while (!q.empty()) {
            int k = q.size();
            for (int i = 0; i < k; i++) {
                pair<int,int> t=q.front();
                q.pop();
                int cx=t.first;
                int cy=t.second;
                if (matrix[cx][cy]== 1)ans[cx][cy]=cntd;
                for (int j = 0; j < 4; j++) {
                    int tx = cx + dx[j];
                    int ty = cy + dy[j];
                    if (tx >= 0 && tx < m && ty >= 0 && ty < n && visit[tx][ty] == 0) {
                        q.push({tx,ty});
                        visit[tx][ty] = 1;
                    }
                }
            }
            cntd++;
        }
        return ans;
    }
};
```
