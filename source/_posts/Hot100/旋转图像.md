# 综述

-   **先转置，然后逐行反转**

# Code

```c++
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n=matrix.size();
        for(int i=0;i<n;i++){//下对角线翻到上对角线
            for(int j=0;j<i;j++){
                swap(matrix[i][j],matrix[j][i]);
            }
        }
        for(int i=0;i<n;i++){//逐行反转
            for(int j=0;j<n/2;j++){
                swap(matrix[i][j],matrix[i][n-j-1]);
            }
        }
    }
};
```
```c++
class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {
        int n=matrix.size();
        for(int i=0;i<n/2;i++)//共有n/2圈
            for(int j=i;j<n-i-1;j++)//每一圈考虑第一个到倒数第二个元素，第一圈是[0,n-1)...=>[i,n-i-1)
            {
                //反转关系是[i,j]=>[j,n-i-1]
                int cur=matrix[i][j],r=j,c=n-i-1;
                for(int i=0;i<4;i++)//进行四次反转,每次循环找出下一次的r和c
                {
                    swap(cur,matrix[r][c]);
                    int tmp=r;
                    r=c;
                    c=n-tmp-1;
                }
            }
    }
};
```

# 复杂度分析

时间复杂度：O(n^2)

空间复杂度：O(1)
