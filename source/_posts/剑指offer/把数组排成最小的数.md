# Code

```c++
class Solution {
public:
    static bool cmp(int a,int b)
    {
        auto as=to_string(a);
        auto bs=to_string(b);
        return as+bs<bs+as;
    }
    string minNumber(vector<int>& nums) {
        sort(nums.begin(),nums.end(),cmp);
        string res;
        for(auto x:nums)res+=to_string(x);
        return res;
    }
};
```
