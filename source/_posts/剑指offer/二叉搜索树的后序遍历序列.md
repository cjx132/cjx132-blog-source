# Code

```c++
class Solution {
public:
    bool verifyPostorder(vector<int>& postorder) {
        return help(postorder,0,postorder.size()-1);
    }

    bool help(vector<int>& postorder, int l,int r)
    {
        if(l>=r)return true;
        int i=l;
        while(postorder[i]<postorder[r])i++;
        int p=i;
        while(postorder[i]>postorder[r])i++;
        return i==r&&help(postorder,l,p-1)&&help(postorder,p,r-1);
    }
};
```
