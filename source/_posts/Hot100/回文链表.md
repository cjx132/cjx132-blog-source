# 综述

**找到中点，反转后一半，开始比较**

# Code

```c++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    bool isPalindrome(ListNode* head) {
        if(!head||!head->next)return true;
        
        ListNode *fast=head,*slow=head;
        while(fast&&fast->next&&fast->next->next)//找到中点，下取整
        {
            slow=slow->next;
            fast=fast->next->next;
        }

        fast=reverse(slow->next);//反转后一半

        slow->next=nullptr;
        
        slow=head;
        while(slow&&fast)//开始比较
        {
            if(slow->val!=fast->val)return false;
            slow=slow->next;
            fast=fast->next;
        }
        return true;
    }

    ListNode* reverse(ListNode *head)
    {
        if(!head||!head->next)return head;
        ListNode *pre=head,*cur=head->next;
        while(cur)
        {
            auto t=cur->next;
            cur->next=pre;
            pre=cur;
            cur=t;
        }
        head->next=nullptr;
        return pre;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(1)
