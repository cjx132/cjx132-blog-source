-   线段树的深度：O(logn)
-   线段树的节点个数：O(n)
-   线段树节点编号方式：线段树数组从下标1开始；节点x的左儿子编号：2x；右儿子编号2x+1
-   节点编号的最大值<=4n
-   原始数组使用都是按照下标左闭右闭

**局限性：**（update方法的局限性）

1.  标记必须可以合并
2.  可以快速更新区间信息

# 递归线段树

## 单点更新，区间查询

```c++
struct Node {
    int data;
    int l;
    int r;
};
Node tree[4005];
int arr[1000];

void build(int x, int l, int r) {
    tree[x].l = l;
    tree[x].r = r;
    if (l == r) {
        tree[x].data = arr[l];
        return;
    }
    int mid = (l + r) / 2;
    build(2 * x, l, mid);
    build(2 * x + 1, mid + 1, r);
    tree[x].data = tree[2 * x].data + tree[2 * x + 1].data;
}

void change(int x, int index, int val) {
    if (tree[x].l == tree[x].r && index == tree[x].l) {
        tree[x].data = val;
        arr[index] = val;
        return;
    }
    int mid = (tree[x].l + tree[x].r) / 2;
    if (index <= mid) {
        change(2 * x, index, val);
    } else {
        change(2 * x + 1, index, val);
    }
    tree[x].data = tree[2 * x].data + tree[2 * x + 1].data;
}

int query(int x, int L, int R) {
    if (tree[x].l == L && tree[x].r == R) {
        return tree[x].data;
    }
    int mid = (tree[x].l + tree[x].r) / 2;
    if (R <= mid) {
        return query(2 * x, L, R);
    } else if (L >= mid + 1) {
        return query(2 * x + 1, L, R);
    } else {
        return query(2 * x, L, mid) + query(2 * x + 1, mid + 1, R);
    }
}
```

## 区间更新，区间查询

不同的题只要改变**update、push\_up的代码**即可解决

```c++
struct Node {
    long l;
    long r;
    long data;
    long lazy_tag;
};
Node tree[400020];
long arr[100005];

void update(long x, long w) {
    long len = tree[x].r - tree[x].l + 1;
    tree[x].data += w * len;
    if (tree[x].lazy_tag == -1)tree[x].lazy_tag = w;
    else tree[x].lazy_tag += w;
}

void push_down(long x) {
    update(2 * x, tree[x].lazy_tag);
    update(2 * x + 1, tree[x].lazy_tag);
    tree[x].lazy_tag = -1;
}

void push_up(long x) {
    tree[x].data = tree[2 * x].data + tree[2 * x + 1].data;
}

void build(long x, long l, long r) {
    tree[x].lazy_tag = -1;
    tree[x].l = l;
    tree[x].r = r;
    if (l == r) {
        tree[x].data = arr[l];
        return;
    }
    long mid = (l + r) / 2;
    build(2 * x, l, mid);
    build(2 * x + 1, mid + 1, r);
    push_up(x);
}

void change(long x, long w, long L, long R) {
    if (tree[x].l == L && tree[x].r == R) {
        update(x, w);
        return;
    }

    if (tree[x].lazy_tag != -1)push_down(x);

    long mid = (tree[x].l + tree[x].r) / 2;
    if (R <= mid) {
        change(2 * x, w, L, R);
    } else if (L >= mid + 1) {
        change(2 * x + 1, w, L, R);
    } else {
        change(2 * x, w, L, mid);
        change(2 * x + 1, w, mid + 1, R);
    }
    push_up(x);
}

long query(long x, long L, long R) {
    if (tree[x].l == L && tree[x].r == R) {
        return tree[x].data;
    }

    if (tree[x].lazy_tag != -1)push_down(x);

    long mid = (tree[x].l + tree[x].r) / 2;
    if (R <= mid) {
        return query(2 * x, L, R);
    } else if (L >= mid + 1) {
        return query(2 * x + 1, L, R);
    } else {
        return query(2 * x, L, mid) + query(2 * x + 1, mid + 1, R);
    }
}
```

## 区间乘法&区间加法&区间修改

需要设计两个懒节点，需要修改push\_down的方法，要注意其中的优先级问题，先乘法后加法

```c++
#include <iostream>

using namespace std;
struct Node {
    long l;
    long r;
    long data;
    //x->x*lazy_tag1+lazy_tag2
    long lazy_tag1;//乘法标记，初始化为1
    long lazy_tag2;//加法标记，初始化为0
};
Node tree[400020];
long arr[100005];
int p;//取模

void update(long x, long w, int type) {
    long len = tree[x].r - tree[x].l + 1;
    if (type == 1) {//乘法
        tree[x].data = tree[x].data * w % p;
        tree[x].lazy_tag1 = tree[x].lazy_tag1 * w % p;
        tree[x].lazy_tag2 = tree[x].lazy_tag2 * w % p;
    } else {//加法
        tree[x].data += w * len;
        tree[x].lazy_tag2 += w;
    }
}

void push_down(long x) {
    //优先级问题，先乘后加
    update(2 * x, tree[x].lazy_tag1, 1);//乘法
    update(2 * x, tree[x].lazy_tag2, 2);//加法
    update(2 * x + 1, tree[x].lazy_tag1, 1);//乘法
    update(2 * x + 1, tree[x].lazy_tag2, 2);//加法
    tree[x].lazy_tag1 = 1;
    tree[x].lazy_tag2 = 0;
}

void push_up(long x) {
    tree[x].data = tree[2 * x].data + tree[2 * x + 1].data;
}

void build(long x, long l, long r) {
    //标记初始化
    tree[x].lazy_tag1 = 1;
    tree[x].lazy_tag2 = 0;
    tree[x].l = l;
    tree[x].r = r;
    if (l == r) {
        tree[x].data = arr[l];
        return;
    }
    long mid = (l + r) / 2;
    build(2 * x, l, mid);
    build(2 * x + 1, mid + 1, r);
    push_up(x);
}

void change(long x, long w, long L, long R, int type) {
    if (tree[x].l == L && tree[x].r == R) {
        update(x, w, type);//1.
        return;
    }

    if (tree[x].lazy_tag1 != 1 || tree[x].lazy_tag2 != 0)push_down(x);//2.

    long mid = (tree[x].l + tree[x].r) / 2;
    if (R <= mid) {
        change(2 * x, w, L, R, type);
    } else if (L >= mid + 1) {
        change(2 * x + 1, w, L, R, type);
    } else {
        change(2 * x, w, L, mid, type);
        change(2 * x + 1, w, mid + 1, R, type);
    }
    push_up(x);//3.
}

long query(long x, long L, long R) {
    if (tree[x].l == L && tree[x].r == R) {
        return tree[x].data;
    }

    if (tree[x].lazy_tag1 != 1 || tree[x].lazy_tag2 != 0)push_down(x);//1.

    long mid = (tree[x].l + tree[x].r) / 2;
    if (R <= mid) {
        return query(2 * x, L, R);
    } else if (L >= mid + 1) {
        return query(2 * x + 1, L, R);
    } else {
        return query(2 * x, L, mid) + query(2 * x + 1, mid + 1, R);//2.
    }
}

int main() {
    long n, m;
    while (cin >> n >> m >> p) {
        for (long i = 0; i < n; i++)cin >> arr[i];
        build(1, 0, n - 1);
        for (long i = 0; i < m; i++) {
            long opt, x, y, k;
            cin >> opt;
            if (opt == 1) {
                cin >> x >> y >> k;
                change(1, k, x - 1, y - 1, 1);
            } else if (opt == 2) {
                cin >> x >> y >> k;
                change(1, k, x - 1, y - 1, 2);
            } else if (opt == 3) {
                cin >> x >> y;
                cout << query(1, x - 1, y - 1) % p << endl;
            }
        }
    }
    return 0;
}
```
