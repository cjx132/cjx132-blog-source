# 综述

-   迭代：由于需要链表前驱节点的信息，因此需要一个变量保存当前节点的前一个节点：pre是cur节点的前一个节点,两个节点依次后移
-   递归写法

# Code

```c++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        if(!head||!head->next)return head;

        ListNode *pre=head;
        ListNode *cur=head->next;

        while(cur)
        {
            ListNode *t=cur->next;
            cur->next=pre;
            pre=cur;
            cur=t;
        }

        head->next=nullptr;
        return pre;
    }
};
```
```c++
class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        if(!head||!head->next)return head;

        auto ans=reverseList(head->next);
        head->next->next=head;
        head->next=nullptr;
        return ans;
    }
};
```
