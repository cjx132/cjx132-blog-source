# Code

```c++
class Solution {
public:
    int res=0;
    int dx[4]={-1,0,1,0},dy[4]={0,1,0,-1};
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int n=grid.size();
        if(n==0)return 0;
        int m=grid[0].size();
        for(int i=0;i<n;i++)
            for(int j=0;j<m;j++)
                dfs(grid,i,j);
        return res;
    }

    int dfs(vector<vector<int>>& grid,int x,int y)
    {
        if(x<0||x>=grid.size()||y<0||y>=grid[0].size()||grid[x][y]==0)
            return 0;

        int ret=1;
        grid[x][y]=0;//注意标记
        for(int i=0;i<4;i++)
        {
            int tx=dx[i]+x,ty=dy[i]+y;
            ret+=dfs(grid,tx,ty);
        }
        res=max(res,ret);
        return ret;
    }
};
```

# 复杂度分析

-   时间复杂度：O(n)
-   空间复杂度：O(1)
