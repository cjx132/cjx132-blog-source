# 综述

-   先找到反转点，然后分别二分搜索

# Code

```c++
class Solution {
public:
    int help(vector<int>& nums,int l,int r,int &target)
    {
        while(l<=r)
        {
            int mid=(l+r)/2;
            if(target==nums[mid])return mid;
            else if(target>nums[mid])l=mid+1;
            else r=mid-1;
        }
        return -1;
    }
    int search(vector<int>& nums, int target) {
        int l=0,r=nums.size()-1;
        while(l<r)
        {
            int mid=(l+r)/2;
            if(nums[mid]<nums[0])r=mid;
            else l=mid+1;
        }
        int t=l;
        return max(help(nums,0,t-1,target),help(nums,t,nums.size()-1,target));
    }
};
```

# 复杂度分析

时间复杂度：O(logn)

空间复杂度：O(1)
