# 题目

输入一棵二叉搜索树，将该二叉搜索树转换成一个排序的循环双向链表。要求不能创建任何新的节点，只能调整树中节点指针的指向。 为了让您更好地理解问题，以下面的二叉搜索树为例：

![img](https://assets.leetcode.com/uploads/2018/10/12/bstdlloriginalbst.png)

我们希望将这个二叉搜索树转化为双向循环链表。链表中的每个节点都有一个前驱和后继指针。对于双向循环链表，第一个节点的前驱是最后一个节点，最后一个节点的后继是第一个节点。

下图展示了上面的二叉搜索树转化成的链表。“head” 表示指向链表中有最小元素的节点。

![img](https://assets.leetcode.com/uploads/2018/10/12/bstdllreturndll.png)

# Code

```c++
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* left;
    Node* right;

    Node() {}

    Node(int _val) {
        val = _val;
        left = NULL;
        right = NULL;
    }

    Node(int _val, Node* _left, Node* _right) {
        val = _val;
        left = _left;
        right = _right;
    }
};
*/
//非递归实现
class Solution {
public:
    Node* treeToDoublyList(Node* root) {
        if(!root)return NULL;
        stack<Node*>st;
        Node* pre=NULL;//当前节点的前驱节点
        Node* head=NULL;//保留头结点
        while(root!=NULL||!st.empty()){
            if(root){
                st.push(root);
                root=root->left;//左子树递归
            }else{
                Node* temp=st.top();
                st.pop();//访问中间节点

                //其他为中序遍历模板，下面为具体逻辑实现
                if(pre==NULL){//只会执行一次，找到最小节点
                    pre=temp;
                    head=temp;
                }else{
                    pre->right=temp;//前驱节点右指针为当前节点
                    temp->left=pre;//当前节点左指针为前驱节点
                    pre=temp;//后移，更新前驱节点
                }

                root=temp->right;//右子树递归
            }
        }
        //首尾节点相连
        pre->right=head;
        head->left=pre;
        return head;
    }
};	

//递归实现
class Solution {
public:
    Node* treeToDoublyList(Node* root) {
        if(!root) return nullptr;
        Node* head = nullptr, *pre = nullptr;
        helper(root, head, pre);
        head->left = pre;
        pre->right = head;
        return head;
    }
    void helper(Node* root, Node*& head, Node*& pre) {
        if(!root)  return;
        helper(root->left, head, pre);
        if(!head) {
            head = root;   // 找到head
            pre = root;    // 对pre进行初始化
        } else {
            pre->right = root;
            root->left = pre;
            pre = root;
        }
        helper(root->right, head, pre);
    }
};
```
