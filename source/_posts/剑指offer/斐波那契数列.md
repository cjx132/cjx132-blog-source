# Code

```c++
#include<bits/stdc++.h>
using namespace std;
long long n;
const int MOD=10000;

void mul(int a[][2],int b[][2],int c[][2])//矩阵乘法,c是答案数组
{
    int tmp[2][2]={{0,0},{0,0}};
    for(int i=0;i<2;i++)
        for(int j=0;j<2;j++)
            for(int k=0;k<2;k++)
            {
                long long x = tmp[i][j] + (long long)a[i][k] * b[k][j];
                tmp[i][j] = x % MOD;
            }
    
    for(int i=0;i<2;i++)
        for(int j=0;j<2;j++)
            c[i][j]=tmp[i][j];

}

void qmi(int a[][2],int k,int res[][2])//a矩阵的k次方，最终答案是res,res初始化为单位矩阵
{
    while(k)
    {
        if(k&1)mul(res,a,res);
        mul(a,a,a);
        k>>=1;
    }
}

int main()
{
    
    while(cin>>n&&n!=-1)
    {
        if(n==0)
        {
            cout<<0<<endl;
            continue;
        }
        
        int a[][2]={{1,1},{1,0}};//关系矩阵
        
        int res[][2]={{1,0},{0,1}};//单位矩阵,最终结果为a^n-1
        qmi(a,n-1,res);
        
        int x1[2]={1,0};//X1,x1[0]为f(1),x1[1]为f[0]
        
        int ans[2]={0,0};//最终结果
        for(int i=0;i<2;i++)
            for(int k=0;k<2;k++)
                {
                    long long r=ans[i]+(long long)res[k][i]*x1[k];
                    ans[i]=r%MOD;
                }
        
        cout<<ans[0]<<endl;
    }
    
    return 0;
}
```

# 复杂度分析

-   时间复杂度：O(k^3logn)
    
-   空间复杂度：O(1)
