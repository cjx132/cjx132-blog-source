# 综述

-   最简单的双指针

# Code

```c++
class Solution {
public:
    //反转一个单词
    void reverse(string &s,int l,int r)
    {
        while(l<=r)
            swap(s[l++],s[r--]);
    }

    string reverseWords(string s) {
        int n=s.size();
        for(int i=0;i<n;i++)//双指针模板，i为单词开始,j为单词结束
        {
            int j=i;
            while(j<n&&s[j]!=' ')j++;
            reverse(s,i,j-1);
            i=j;//循环中还有一步i++,因此这里是i=j
        }
        return s;
    }
};
```
