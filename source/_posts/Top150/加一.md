# Code

```c++
class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        for(int i=digits.size()-1;i>=0;i--){
            digits[i]++;
            digits[i]=digits[i]%10;
            if(digits[i])return digits;
        }
        //999+1=1000(首位是1，其余均为0)
        digits=vector<int>(digits.size()+1,0);
        digits[0]=1;
        return digits;
    }
};
```
