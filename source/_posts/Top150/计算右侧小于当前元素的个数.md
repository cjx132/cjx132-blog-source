## 归并排序解法

**此题是从大到小排序，体会和数组中的逆序对中的区别**

```c++
class Solution {
    typedef pair<int,int>PII;
public:
    vector<PII>nums,tmp;
    vector<int>ans;
    void merge_sort(int l,int r)
    {
        if(l>=r)return;
        int mid=(l+r)>>1;
        merge_sort(l,mid),merge_sort(mid+1,r);
        int i=l,j=mid+1,k=0;
        while(i<=mid&&j<=r)
        {
            if(nums[i].first>nums[j].first)
            {
                ans[nums[i].second]+=r-j+1;
                tmp[k++]=nums[i++];
            }
            else
                tmp[k++]=nums[j++];
        }
        while(i<=mid)tmp[k++]=nums[i++];
        while(j<=r)tmp[k++]=nums[j++];
        
        for(int i=l,j=0;i<=r;i++,j++)nums[i]=tmp[j];
    }
    vector<int> countSmaller(vector<int>& _nums) {
        int n=_nums.size();
        tmp=vector<PII>(n,{-1,-1});
        ans=vector<int>(n,0);
        for(int i=0;i<n;i++)
            nums.push_back({_nums[i],i});
        merge_sort(0,n-1);
        return ans;
    }
};
```

## 树状数组解法

```java
class Solution {
    int n;
    int M = (int) (10e5 + 2);
    int[] A = new int[M];//离散之后得到的原数组:A数组长度为n，数组内元素范围是[1,n]，A数组中的值存储对应nums数组元素的排名
    int[] C = new int[M];//树状数组

    public List<Integer> countSmaller(int[] nums) {
        List<Integer> counts = new LinkedList<>();
        n = nums.length;

        //以下代码实现的操作相当于：求解给定N个学生的分数，给他们进行排名，分数相同则排名相同
        //最终得到离散之后的数组A：A数组长度为n，数组内元素范围是[1,n]，A数组中的值存储对应nums数组元素的排名
        Node[] temps = new Node[n];//结构体数组
        for (int i = 0; i < n; i++) {
            temps[i] = new Node();
            temps[i].setPos(i);
            temps[i].setVal(nums[i]);
        }

        //按照val排序
        Arrays.sort(temps, (a, b) -> {
            return a.val - b.val;
        });

        for (int i = 0; i < temps.length; i++) {
            if (i == 0 || temps[i].val != temps[i - 1].val) {
                A[temps[i].pos] = i + 1;
            } else {
                A[temps[i].pos] = A[temps[i - 1].pos];
            }
        }

        //求解本题的关键代码
        //由于是求右边小的元素：所以倒序遍历；并且最终结果每次从头部插入

        for (int i = nums.length - 1; i >= 0; i--) {
            //这里想象一个虚拟的hash数组，这个数组的长度和树状数组C大小相同为n+1；这个数组的下标是A[i]的值，这个数组中的值是A[i]出现的次数
            //树状数组所求的前缀和其实就是这个hash数组的前缀和

            add(A[i], 1);//给虚拟hash数组下标为A[i]的地方+1，同时更新树状数组的值
            counts.add(0, getSum(A[i] - 1));//求虚拟hash数组下标范围[1,A[i]-1]的前缀和
        }
        return counts;
    }

    //树状数组getSum函数模板
    //返回前index个整数之和
    public int getSum(int index) {
        int sum = 0;
        for (int i = index; i > 0; i -= i & (-i)) {//注意是i>0而不是i>=0
            sum += C[i];
        }
        return sum;
    }

    //树状数组add函数模板
    //add函数将第index个整数加上value:单点更新
    public void add(int index, int value) {
        for (int i = index; i <= n; i += i & (-i)) {//注意i必须能取到n
            C[i] += value;
        }
    }

}
class Node {
    public int val;//序列元素的值
    public int pos;//原始序号

    public int getVal() {
        return val;
    }

    public void setVal(int val) {
        this.val = val;
    }

    public int getPos() {
        return pos;
    }

    public void setPos(int pos) {
        this.pos = pos;
    }
}
```
