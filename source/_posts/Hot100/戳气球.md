# 综述

-   以现在我的理解，再来说说这道题目。  
    这道题目属于一道经典的 区间DP 的题目，在计算这种问题时，通常第一维度是从小到大来枚举区间的长度（规模）；第二维度枚举一个 k 来计算当前长度（规模）的值。  
    如果从这个角度来理解的话，我们应该不难想到状态空间的表示可以用一个二维的数组来表示了。dp`[i][j]`，i 代表区间的上限，j 代表区间的下限。
-   [回溯+记忆化递归+区间dp](https://qoogle.top/leetcode-312-burst-balloons/)

```c++
class Solution {
public:
    /*记忆化递归
    vector<vector<int> >dp;
    int help(vector<int>& nums, int i,int j){
        if(i>j)return 0;
        if(dp[i][j]>0)return dp[i][j];
        for(int k=i;k<=j;k++){
            int left=help(nums,i,k-1);
            int right=help(nums,k+1,j);
            int t=nums[k]*nums[i-1]*nums[j+1];
            dp[i][j]=max(dp[i][j],left+t+right);

        return dp[i][j];

       dp = vector<vector<int> >(n + 2, vector<int>(n + 2, 0));

       nums.push_back(1);
       return help(nums,1,n);
    }
    */
    int maxCoins(vector<int>& nums) {
       int n=nums.size();
       vector<vector<int> >dp(n + 2, vector<int>(n + 2, 0));
       nums.insert(nums.begin(), 1);
       nums.push_back(1);
       for(int len=1;len<=n;len++){
           for(int i=1;i<=n-len+1;i++){
               int j=i+len-1;
               for(int k=i;k<=j;k++){
                   dp[i][j]=max(dp[i][j],dp[i][k-1]+nums[k]*nums[i-1]*nums[j+1]+dp[k+1][j]);
               }
           }
       }
       return dp[1][n];
    }
};
```
