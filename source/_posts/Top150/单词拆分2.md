# 综述

1.  首先想到使用字典树：在回溯中，根据字典树前缀进行剪枝，但是这个剪枝的判断时间复杂度不是O(1)，仍然超时。
    
2.  > -   结合[单词拆分](https://cjx132.github.io/2020/08/14/Hot100/%E5%8D%95%E8%AF%8D%E6%8B%86%E5%88%86/)中的dp思路进行剪枝：dp\[i\]表示字符串s，以下标i开始到最后的字符串是否可被 wordDict 拆分
    > -   使用哈希表实现，剪枝的时间复杂度是O(1)
    
3.  关于字典树以及哈希表的一些思考：
    
    > -   HashMap 实现的是哈希表，用于解决O(1)的精确查找，无论是内存中实现程序逻辑还是外存中实现 key - value 存储，几乎无处不用
    > -   Trie 树即字典树，用于解决前缀检索（模糊查找），最经典的应用就是搜索 suggest，搜索”变形”，suggest 给你”变形金刚”
    > -   从 Trie 当然也能演化出后缀树，解决后缀检索问题
    > -   **Trie 能解决 HashMap 的问题（精确检索本来就是模糊检索的一个特例），只不过速度没 HashMap 快**
    > -   HashMap 不能解决 Trie 的问题
    

# Code

## 使用前缀树，但是由于剪枝的判断时间复杂度不是O(1)仍然**超时**

```c++
class Trie{
    public:
    bool isEnd;
    Trie* next[26];

    void insert(string str){
        Trie* node=this;
        for(char c:str){
            if(node->next[c-'a']==nullptr){
                node->next[c-'a']=new Trie();
            }
            node=node->next[c-'a'];
        }
        node->isEnd=true;
    }
};
class Solution {
public:
    vector<vector<string> >res;
    vector<string>ss;
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        Trie* node=new Trie();
        for(string str:wordDict){
            node->insert(str);
        }
        dfs(s,node,0);
        vector<string>ans;
        for(auto item:res){
            string temp="";
            for(string sitem:item){
                temp+=sitem;
                temp+=" ";
            }
            ans.push_back(temp.substr(0,temp.size()-1));
        }
        return ans;
    }
    void dfs(string& s,Trie*& node,int left){
        if(s.size()==left){
            res.push_back(ss);
            return;
        }
        for(int i=left;i<s.size();i++){
            string temp=s.substr(left,i-left+1);
            Trie* root=node;
            for(int j=0;j<temp.size();j++){
                root=root->next[temp[j]-'a'];
                if(root==nullptr){
                    return;
                }
            }
            if(root->isEnd==true){
                ss.push_back(temp);
                dfs(s,node,i+1);
                ss.pop_back();
            }
        }
    }
};
```

## dp剪枝

```c++
class Solution {
public:
    vector<vector<string> >res;
    vector<string>ss;
    vector<string> wordBreak(string s, vector<string>& wordDict) {
        unordered_set<string>se(wordDict.begin(),wordDict.end());
        vector<int>dp(s.size()+5,0);

        dp_help(s,wordDict,dp,se);

        dfs(s,0,dp,se);

        //结果处理
        vector<string>ans;
        for(auto item:res){
            string temp="";
            for(string sitem:item){
                temp+=sitem;
                temp+=" ";
            }
            ans.push_back(temp.substr(0,temp.size()-1));
        }
        return ans;
    }

    void dp_help(string s, vector<string>& wordDict, vector<int>& dp,unordered_set<string>& se){
        //单词搜索中的解法，注意这里dp的含义
        dp[s.size()]=1;
        for(int i=s.size()-1;i>=0;i--){
            for(int j=s.size();j>i;j--){
                if(dp[j]&&se.count(s.substr(i,j-i))){
                    dp[i]=true;
                    break;
                }
            }
        }
    }

    void dfs(string& s,int left,vector<int>& dp,unordered_set<string>& se){
        if(s.size()==left){
            res.push_back(ss);
            return;
        }

        if(!dp[left])return;//利用单词拆分中的dp进行剪枝

        for(int i=left;i<s.size();i++){
            string temp=s.substr(left,i-left+1);
            if(!se.count(temp))continue;//不在字典中，剪枝
            ss.push_back(temp);
            dfs(s,i+1,dp,se);
            ss.pop_back();
        }
    }
};
```
