# Code

```c++
class Solution {
public:
    int majorityElement(vector<int>& nums) {
        int res=0,count=0;
        for(auto x:nums)
        {
            if(count==0)
            {
                res=x;
                count++;
            }
            else if(res==x)count++;
            else count--;
        }
        return res;
    }
    
};
```
