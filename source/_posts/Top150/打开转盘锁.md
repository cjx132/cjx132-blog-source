```c++
class Solution {
public:
    int openLock(vector<string>& deadends, string target) {
        unordered_set<string>deadendsSet(deadends.begin(),deadends.end());//字典存入set，加快查找速度
        unordered_set<string>visit;//标记数组，标记是否已被访问

        unordered_set<string>q1;//从前往后扩散时，每层保存的节点
        unordered_set<string>q2;//从后往前扩散时，每层保存的节点

        //注意点1：必须进行特判，因为从两端往中间扩散时，可能会存在这种情况：
        //target根本不在字典中，在往前扩散时，可能会存在公共交集的情况
        if(deadendsSet.count(target)!=0||deadendsSet.count("0000")){
            return -1;
        }

        q1.insert("0000");
        q2.insert(target);
        int ans=-1;
        while(!q1.empty()){
            ans++;//层数+1
            unordered_set<string>temp;

            for(auto eachstr:q1){//一次遍历一层
                visit.insert(eachstr);//注意点2：visit数组更新的位置 
                if(q2.count(eachstr)!=0)return ans;//存在交集，满足条件即可返回
                
                for(int i=0;i<4;i++){
                    string str=eachstr;//不可少，保证每次都是基于eachstr的值修改
                    for(int j=0;j<2;j++){
                        str[i]=getChar(eachstr,i,j);
                        if(deadendsSet.count(str)==0&&visit.count(str)==0){
                            temp.insert(str);
                            //和单源bfs不同，绝对不能在这里更新visit
                        }  
                    }
                }
            }
            //正常情况下，下次访问q2
            q1=q2;
            q2=temp;

            //优化：从集合元素少的扩散
            if(q1.size()>q2.size()){
                swap(q1,q2);
            }
        }
        return -1;
    }

    char getChar(string s,int i,int k){
        char c=s[i];
        if(k==0){//增加
            if(c=='9')return '0';
            else return c+1;
        }else{
            if(c=='0')return '9';
            else return c-1;
        }
    }
};
```
