```c++
//dfs解法，超时
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m=grid.size();
        if(m==0)return 0;
        int n=grid[0].size();
        int ans=INT_MAX;
        dfs(grid,m,n,0,0,ans,grid[0][0]);
        return ans;
    }
    void dfs(vector<vector<int>>& grid,int m,int n,int x,int y,int &ans,int sum){
        if(x>=m||y>=n)return;
        if(x==m-1&&y==n-1){
            if(sum<ans){
                ans=sum;
            }
            return;
        }
        if(x+1<m&&sum+grid[x+1][y]<ans){
            sum+=grid[x+1][y];
            dfs(grid,m,n,x+1,y,ans,sum);
            sum-=grid[x+1][y];
        }
        if(y+1<n&&sum+grid[x][y+1]<ans){
            sum+=grid[x][y+1];
            dfs(grid,m,n,x,y+1,ans,sum);
            sum-=grid[x][y+1];
        }
    }
};
//动态规划解法
class Solution {
public:
    int minPathSum(vector<vector<int>>& grid) {
        int m=grid.size();
        if(m==0)return 0;
        int n=grid[0].size();
        int ans=INT_MAX;
        vector<vector<int> >dp(m+1,vector<int>(n+1,INT_MAX));
        dp[0][0]=grid[0][0];
        for(int i=1;i<m;i++)dp[i][0]=dp[i-1][0]+grid[i][0];
        for(int j=1;j<n;j++)dp[0][j]=dp[0][j-1]+grid[0][j];
        for(int i=1;i<m;i++){
            for(int j=1;j<n;j++){
                dp[i][j]=min(dp[i-1][j],dp[i][j-1])+grid[i][j];
            }
        }
        return dp[m-1][n-1];
    }
};
```
