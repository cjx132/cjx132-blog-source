# Code

```c++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
vector<int>res;
    vector<int> reversePrint(ListNode* head) {
        //递归写法
        // if(head==NULL){
        //     return res;
        // }
        // reversePrint(head->next);
        // res.push_back(head->val);
        // return res;
        
        //反转链表
        ListNode* pre=NULL;
        ListNode* cur=head;
        ListNode* next=head;
        while(cur){
            next=cur->next;//反转前保存值
            cur->next=pre;//反转链表
            pre=cur;//后移
            cur=next;//后移
        }
        while(pre){//此时pre节点为头结点
        res.push_back(pre->val);
        pre=pre->next;
        }
        return res;
    }
};
```
