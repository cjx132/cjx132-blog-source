# Code

```plain
class Solution {
public:
    int majorityElement(vector<int>& nums) {
       //超过一半的数字比其他所有数字的总和次数多
        int n=0;
        int res;
        int len=nums.size();
        for(int i=0;i<len;i++){
            if(n==0) res=nums[i];//抵消后赋予新值
            if(res==nums[i]) n++;
            else n--;
        }
        return res;
    }
};
```
