# Code

```plain
class Solution {
public:
    int findNthDigit(int n) {
        if(n<10)return n;
        int i=0;//几位数
        int num=0;//几位数中的第几个字符
        for(;n>=0;i++){
            num=n;//保留上一个n值
            n-=pow(10,i)*9*(i+1);
        }
        int a=(num-1)/i;//几位数中的第几个数
        int b=(num-1)%i;//所求结果在这个数中是第几位
        a=pow(10,i-1)+a;//这个数的值
        string s=to_string(a);
        return (int)(s[b]-'0');
    }
};
```
