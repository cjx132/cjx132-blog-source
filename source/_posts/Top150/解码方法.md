    **逻辑是：先把当前是不是0优先判断一下，以后根据前一个字符的值确定判断的逻辑**
    

# Code

```c++
class Solution {
public:
    int numDecodings(string s) {
        int n=s.size();
        vector<int>dp(n+1,0);
        if(s[0]=='0')return 0;
        dp[0]=1;
        dp[1]=1;
        for(int i=1;i<n;i++)
        {
            if(s[i]=='0')
            {
                if(s[i-1]=='1'||s[i-1]=='2')dp[i+1]=dp[i-1];
                else return 0;
            }
            else if(s[i-1]=='1')
            {
                dp[i+1]=dp[i]+dp[i-1];
            }
            else if(s[i-1]=='2')
            {
                if(s[i]>='1'&&s[i]<='6')dp[i+1]=dp[i]+dp[i-1];
                else dp[i+1]=dp[i];
            }
            else 
            {
                dp[i+1]=dp[i];
            }
        }
        return dp[n];
    }
};
```

```c++
class Solution {
public:
    int numDecodings(string s) {
        int n=s.size();
        if(s[0]=='0')return 0;//加一个特判
        // vector<int>dp(n+5,0);
        int pre=1;
        int cur=1;
        // dp[0]=1;
        // dp[1]=1;
        // dp[i+1]代表s[0,i]中的解码方法总数
        for(int i=1;i<n;i++){
            if(s[i]=='0'){//为0的特殊处理
                if(s[i-1]=='1'||s[i-1]=='2'){
                    // dp[i+1]=dp[i-1];
                    int temp=cur;
                    cur=pre;
                    pre=temp;
                }else{
                    return 0;
                }
            }else{//不为0才这样处理
                if(s[i-1]=='1'||(s[i-1]=='2'&&s[i]>='1'&&s[i]<='6')){
                    // dp[i+1]=dp[i]+dp[i-1];
                    int temp=cur;
                    cur=cur+pre;
                    pre=temp;
                }else{
                    int temp=cur;
                    cur=cur;
                    pre=temp;
                    // dp[i+1]=dp[i];
                }
            }
        }
        return cur;
    }
};
```

# 复杂度分析

-   时间复杂度：O(n)
-   空间复杂度：O(1)
