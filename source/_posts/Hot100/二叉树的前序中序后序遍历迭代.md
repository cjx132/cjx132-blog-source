# Code

-   **中序迭代遍历**

```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> inorderTraversal(TreeNode* root) {
        vector<int>res;
        stack<TreeNode*>st;
        while(root||!st.empty())
        {
            if(root)
            {
                st.push(root);
                root=root->left;
            }
            else
            {
                auto t=st.top();
                res.push_back(t->val);
                st.pop();
                root=t->right;
            }
        }
        return res;
    }
};
```

-   **后序迭代遍历**
    
    先遍历左子树，再遍历右子树
    

```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> postorderTraversal(TreeNode* root) {
        stack<TreeNode*>st;
        vector<int>res;
        TreeNode *prev=nullptr;//最后从右子树依次向上返回时用于标记右子树已经被访问过
        while(!st.empty()||root)
        {
            while(root)
            {
                st.push(root);
                root=root->left;
            }
            root=st.top();
            if(!root->right||root->right==prev)//条件成立说明右子树已经遍历过
            {
                res.push_back(root->val);
                st.pop();
                prev=root;//标记当前节点被访问过
                root=nullptr;//根节点已经被访问，说明整个子树已经遍历完成，设置为空，下次循环弹出下一个栈顶元素
            }
            else
                root=root->right;
        }
        return res;
    }
};
```

前序遍历：根左右

变形：根右左

后续遍历：反转上面的

```c++
class Solution {
public:
    vector<int>ans;
    vector<int> postorderTraversal(TreeNode* root) {
        if(!root)return ans;
        stack<TreeNode*>st;

        while(root||st.size())
        {
            if(root)
            {
                ans.push_back(root->val);
                st.push(root);
                root=root->right;
            }
            else
            {
                auto t=st.top();
                st.pop();
                root=t->left;
            }
        }
        reverse(ans.begin(),ans.end());
        return ans;
    }
};
```
