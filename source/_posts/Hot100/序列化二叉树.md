# Code

```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Codec {
public:

    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        ostringstream out;
        queue<TreeNode*>q;
        q.push(root);
        while(q.size())
        {
            auto t=q.front();
            q.pop();
            if(t)
            {
                out<<to_string(t->val)+" ";
                q.push(t->left);
                q.push(t->right);
            }
            else out<<"null ";
        }
        cout<<out.str();
        return out.str();//注意用法
        //对于所给样例，最终返回的字符串如下（最后存在一个空格）：
        //"1 2 3 null null 4 5 null null null null "
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        vector<TreeNode*>q;
        istringstream input(data);//注意用法
        string str;
        while(input>>str)
        {
            if(str=="null")q.push_back(nullptr);
            else q.push_back(new TreeNode(stoi(str)));
        }
        int i=0,j=1;//i为当前节点下标,j为当前节点左子树根节点的下标
        for(;j<q.size();i++)//判断条件是j，因为j先到达结尾
        {
            //为空时直接跳过，不需要j++的原因是序列化时，为空的时候没有对其左右子树序列化，导致少了2位
            //因此后面的节点的下标将前移2位，这样到最后正好遍历完。
            if(!q[i])continue;
            q[i]->left=q[j++];
            q[i]->right=q[j++];
        }
        return q[0];
    }
};
```
