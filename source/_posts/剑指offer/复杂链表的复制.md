# Code

```c++
/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        for(auto p=head;p;)//1.在原链表每个节点末尾复制一个拼上
        {
            Node *np=new Node(p->val);
            auto tmp=p->next;
            p->next=np;
            np->next=tmp;
            p=tmp;
        }   

        for(auto p=head;p;p=p->next->next)//2.赋值random指针
            if(p->random)
                p->next->random=p->random->next;

        Node *dummy=new Node(-1);
        auto cur=dummy;
        for(auto p=head;p;p=p->next)//3.提取出新的链表并还原原链表
        {
            cur->next=p->next;
            cur=cur->next;
            p->next=p->next->next;//还原原链表
        }

        return dummy->next;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(1)
