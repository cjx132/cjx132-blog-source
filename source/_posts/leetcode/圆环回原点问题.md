# Code

```c++
#include<bits/stdc++.h>
using namespace std;
const int N=100;
int f[N][10];
int main()
{
    int n;
    cin>>n;
    f[0][0]=1;
    for(int i=1;i<=n;i++)
        for(int j=0;j<10;j++)
            f[i][j]=f[i-1][(j-1+10)%10]+f[i-1][(j+1)%10];
    cout<<f[n][0]<<endl;
    return 0;
}
```
