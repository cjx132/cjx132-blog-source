# Code

```c++
class Solution {
public:
    vector<int> printNumbers(int n) {
        vector<int>res;
        if(n==0)return res;
        //求10^n,利用快速幂
        long ans=1;//快速幂的结果
        int a=10;//进位
        while(n){
            if(n&1==1){//n的最后一位是1
                ans*=a;
            }
            n>>=1;
            a*=a;
        }
        for(int i=1;i<ans;i++)res.push_back(i);
        return res;
    }
};
```
