# Code

```c++
class Solution {
public:
    vector<int> twoSum(vector<int>& nums, int target) {
        unordered_map<int,int>h;
        vector<int>ans;
        int n=nums.size();
        for(int i=0;i<n;i++)
        {
            if(h.count(target-nums[i]))
            {
                ans.push_back(h[target-nums[i]]);
                ans.push_back(i);
                return ans;
            }
            else h[nums[i]]=i;
        }
        return ans;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(n)
