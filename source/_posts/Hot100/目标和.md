-   **如果是一个数组中挑几个数字和为目标和，这种题目可以直接用01背包动态规划求解。**

# Code

```c++
//暴力超时
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int S) {
        return dfs(nums,S,0);
    }
    int dfs(vector<int>& nums, long target,int k){
        if(target==0&&k==nums.size())return 1;
        if(k>=nums.size())return 0;
        int ans=0;
        ans+=dfs(nums,target-nums[k],k+1);
        ans+=dfs(nums,target+nums[k],k+1);
        return ans;
    }
};
```
```c++
//转化为0-1背包问题
class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int S) {
        //和有可能超出整数范围
        long sum=0;
        for(int num:nums){
            sum+=num;
        }
        //target的2倍不可能为奇数
        if((sum+S)&1)return 0;
        //总和小于S直接返回
        if(S>sum)return 0;
        int target=(sum+S)/2;
        //注意dp数组的初始化
        vector<long>dp(target+1,0);
        dp[0]=1;
        for(int num :nums){
            for(int j=target;j>=num;j--){
                dp[j]+=dp[j-num];
            }
        }
        return dp[target];
    }
};
```

# 复杂度分析

时间复杂度：O(n\*target)

空间复杂度：O(target)
