# Code

```c++
class Solution {
public:
    int max_(int a,int b,int c){
        return max(max(a,b),c);
    }
    int min_(int a,int b,int c){
        return min(min(a,b),c);
    }
    int maxProduct(vector<int>& nums) {
        int len=nums.size();
        int dpmax=nums[0];
        int dpmin=nums[0];
        int ans=nums[0];
        for(int i=1;i<len;i++){
            int temp=dpmax;//更新dpmin时需要dpmax的前一个值
            dpmax=max_(dpmax*nums[i],dpmin*nums[i],nums[i]);
            dpmin=min_(temp*nums[i],dpmin*nums[i],nums[i]);
            ans=max(ans,dpmax);
        }
        return ans;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(1)
