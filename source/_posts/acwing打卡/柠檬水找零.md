# 综述

先找大的，维护两个变量记录5元和10元的个数

# Code

```c++
class Solution {
public:
    bool lemonadeChange(vector<int>& bills) {
        int fives=0,tens=0;
        for(int i=0;i<bills.size();i++){
            if(bills[i]==5){
                fives++;
            }
            else if(bills[i]==10){
                if(fives){
                    fives--;
                    tens++;
                }
                else return false;
            }else {
                int t=15;
                if(tens){
                    tens--;
                    t-=10;
                }
                while(t&&fives){
                    t-=5;
                    fives--;
                }
                if(t)return false;
            }
        }
        return true;
    }
};
```
