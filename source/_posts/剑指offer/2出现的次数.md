# Code

```c++
class Solution {
public:
    const int N=15;
    
    long long f[15][10];

    int ctr[15];
    void init(){
        for(int i=0;i<=9;i++){
            f[1][i]=i==2;
        }
        long long power=10;
        for(int i=2;i<N;i++,power*=10){
            for(int j=0;j<=9;j++){
                for(int k=0;k<=9;k++){
                    f[i][j]+=f[i-1][k];
                }
                if(j==2)f[i][j]+=power;
            }
        }
    }
    int dp(int n) {
        if (!n)return 0;

        int index = 0;
        ctr[index++] = 1;
        long long w=1;
        vector<int> nums;
        while (n) {
            ctr[index] =  n%10*w + ctr[index-1] ;
            w*=10;
            index++;
            nums.push_back(n % 10);
            n /= 10;
        }

        int res = 0;
        int last = 0;
        for (int i = nums.size() - 1; i >= 0; i--) {
            int x = nums[i];
            for (int j = 0; j < x; j++) {
                res += f[i + 1][j];
            }
            if (last == 2)res += ctr[i + 1];
            last = x;
            if(!i&&last==2)res++;
        }
        return res;
    }
    int numberOf2sInRange(int n) {
        init();
        return dp(n)-dp(0);
    }
};
```
