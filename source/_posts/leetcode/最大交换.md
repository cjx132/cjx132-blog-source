# 综述

-   从左往右对于每一个当前数字->右边的比它大的要交换到前面来->如果有多个比它大的要找到最大的->如果最大的有多个，要拿最后一个和它交换
-   `last[num]`数组记录数字num出现的最后位置，初始化为-1，表示没有出现过;

# Code

```c++
class Solution {
public:
    int maximumSwap(int num) {
        string str=to_string(num);
        int n=str.size();

        int last[10];//记录每个数字出现的最后的位置，初始化为-1，表示没有出现过;
        memset(last,-1,sizeof last);

        for(int i=0;i<n;i++)
            last[str[i]-'0']=i;
        
        for(int i=0;i<n;i++)
            for(int j=9;j>str[i]-'0';j--)//数字从大到小枚举
                if(last[j]>i)
                {
                    swap(str[i],str[last[j]]);
                    return stoi(str);
                }

        return num;
    }
};
```

# 复杂度分析

-   时间复杂度：O(n)
-   空间复杂度：O(1)
