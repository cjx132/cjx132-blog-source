**只有加法不用去除前导0**

# 高精度加法

```c++
// C = A + B, A >= 0, B >= 0
vector<int> add(vector<int> &A, vector<int> &B)
{
    vector<int> C;
    for(int i=0,t=0;i<A.size()||i<B.size()||t;i++)
    {
        if(i<A.size())t+=A[i];
        if(i<B.size())t+=B[i];
        C.push_back(t%10);
        t/=10;
    }
    return C;
}
```

# 高精度减法

```c++
// C = A - B, 满足A >= B, A >= 0, B >= 0
vector<int> sub(vector<int> &A, vector<int> &B)
{
    vector<int> C;
    for (int i = 0, t = 0; i < A.size(); i ++ )
    {
        t = A[i] - t;
        if (i < B.size()) t -= B[i];
        C.push_back((t + 10) % 10);
        if (t < 0) t = 1;
        else t = 0;
    }

    while (C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}
```

# 高精度乘法

##大整数A乘小整数b

1.  需要特判最后的进位,且t必须写进循环里
2.  需要去除前导0
    ```c++
    vector<int> mul(vector<int>& A,int b)
    {
        vector<int>C;
        int t=0;
        for(int i=0;i<A.size()||t;i++)
        {
            if(i<A.size())t=A[i]*b+t;
            C.push_back(t%10);
            t/=10;
        }
        while(C.size()>1&&C.back()==0)C.pop_back();
        return C;
    }
    ```
    ##大整数乘大整数
3.  从低位到高位计算，依次确定C数组的每一位
4.  需要去除前导0
    ```c++
    vector<int>mul(vector<int> &A,vector<int> &B)
    {
        vector<int>C(A.size()+B.size(),0);
        for(int i=0;i<A.size();i++)
            for(int j=0;j<B.size();j++)
            {
                int p1=i+j,p2=i+j+1;
                int t=A[i]*B[j]+C[p1];
                C[p1]=t%10; //该位置的值已经最终确定，取余即可
                C[p2]+=t/10;//该位置的值还未确定，需要累加进位
            }
        while(C.size()>1&&C.back()==0)C.pop_back();
        return C;
    }
    ```
    

# 高精度除以低精度

```c++
// A / b = C ... r, A >= 0, b > 0
vector<int> div(vector<int> &A, int b, int &r)
{
    vector<int> C;
    r = 0;
    for (int i = A.size() - 1; i >= 0; i -- )
    {
        r = r * 10 + A[i];
        C.push_back(r / b);
        r %= b;
    }
    reverse(C.begin(), C.end());
    while (C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}
```
