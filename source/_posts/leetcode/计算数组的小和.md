# 综述

-   数组中的逆序对问题
-   从小到大排序
-   在merge时，左段数组L和右段数组R都是有序的了。当L\[i\]<=R\[j\]时，表示L\[i\]比R\[j\]~R\[r\]的元素都要小。因此res需加上j及以后元素的个数\*L\[i\]，即res+=(r-j+1) \* L\[i\]。

# Code

```c++
# include <bits/stdc++.h>
using namespace std;
int q[100010], tmp[100010];

long long merge_sort(int l, int r)
{
    if(l >= r)return 0;
    int mid = l + (r - l) / 2;
    long long res=merge_sort(l, mid) + merge_sort(mid+1, r);
    int i=l, j=mid+1, k=0;
    while(i <= mid && j <= r)
    {
        if(q[i] <= q[j])
        {
            res += q[i] * (r-j+1);
            tmp[k++] = q[i++];
        }
        else tmp[k++] = q[j++];
    }
    while(i <= mid)tmp[k++] = q[i++];
    while(j <= r)tmp[k++] = q[j++];
    for(int i = l, k=0; i <= r; i++, k++)q[i] = tmp[k];
    return res;
}
int main()
{
    int n;
    cin >> n;
    for (int i=0; i < n; i++)
        cin >> q[i];
    
    cout << merge_sort(0, n-1) << endl;
    return 0;
}
```
