# Code

```c++
class Solution {
public:
    int f[1010][1010];
    int findLength(vector<int>& A, vector<int>& B) {
        int res=0;
        for(int i=1;i<=A.size();i++)
            for(int j=1;j<=B.size();j++)
            {
                if(A[i-1]==B[j-1])
                    f[i][j]=f[i-1][j-1]+1;
                else f[i][j]=0;
                res=max(res,f[i][j]);
            }
                
        return res;
    }
};
```
