# Code

```plain
class Solution {
public:
    bool validateStackSequences(vector<int>& pushed, vector<int>& popped) {
        int m=pushed.size();
        int n=popped.size();
        if(m!=n)return false;
        stack<int>s;
        int i=0;
        int j=0;
        for(i=0;i<m;i++){
            s.push(pushed[i]);
            while(j<m&&!s.empty()&&s.top()==popped[j]){
                s.pop();
                j++;
            }
        }
        return s.empty();
    }
};
```
