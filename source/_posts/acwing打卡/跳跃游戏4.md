# Code

```c++
class Solution {
public:
    int maxResult(vector<int>& nums, int k) {
        int n=nums.size();
        vector<int>dp(n+5,0);
        dp[1]=nums[0];

        deque<pair<int,int>>dq;
        dq.push_back(make_pair(1,nums[0]));
        
        for(int i=2;i<=n;i++){
            while(!dq.empty()&&dq.front().first<i-k)dq.pop_front();
            dp[i]=nums[i-1]+dq.front().second;
            while(!dq.empty()&&dq.back().second<=dp[i])dq.pop_back();
            dq.push_back(make_pair(i,dp[i]));
        }
        return dp[n];
    }
};
```
