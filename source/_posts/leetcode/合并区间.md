# Code

```c++
class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {
        vector<vector<int>>ans;

        sort(intervals.begin(),intervals.end());
        int st=-2e9,ed=-2e9;
        for(auto item:intervals)
        {
            if(item[0]>ed)
            {
                if(st!=-2e9)
                    ans.push_back({st,ed});
                st=item[0],ed=item[1];
            }
            else ed=max(ed,item[1]);
        }
        if(st!=-2e9)ans.push_back({st,ed});
        return ans;
    }
};
```
