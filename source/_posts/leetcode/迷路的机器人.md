# Code

```c++
class Solution {
public:
    vector<vector<int>>ans;
    vector<vector<int>> pathWithObstacles(vector<vector<int>>& obstacleGrid) {
        int n=obstacleGrid.size();
        int m=obstacleGrid[0].size();
        int f[105][105]={0};
        if(obstacleGrid[0][0]==0)f[0][0]=1;
        for(int i=1;i<n;i++)f[i][0]=obstacleGrid[i][0]==1?0:f[i-1][0];
        for(int j=1;j<m;j++)f[0][j]=obstacleGrid[0][j]==1?0:f[0][j-1];
        for (int i=1;i<n;i++)
            for (int j=1;j<m;j++)
                {
                    if(obstacleGrid[i][j]==1)
                        f[i][j]=0;
                    else f[i][j]=f[i-1][j]||f[i][j-1];
                }

        int i=n-1,j=m-1;
        if(f[i][j])ans.push_back({i,j});
        else return ans;
        while(i>0||j>0)
        {
            cout<<i<<' '<<j<<endl;
            if(i-1>=0&&f[i-1][j]==1)
            {
                i--;
                ans.push_back({i,j});
            }
            else
            {
                j--;
                ans.push_back({i,j});
            }
        }
        reverse(ans.begin(),ans.end());
        return ans;
    }
};
```
