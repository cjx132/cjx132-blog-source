# Code

```c++
class Solution {
public:
    string toHex(int num) {
        if(num==0)return "0";//特判
        string str="0123456789abcdef";

        string res;
        unsigned int n=num;
        while(n>0)
        {
            res=str[n&15]+res;//掩码
            n>>=4;
        }
        return res;
    }
};
```
