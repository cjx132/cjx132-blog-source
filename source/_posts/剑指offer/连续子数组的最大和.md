# 综述

`dp[i]`表示以`nums[i]`**结尾**的最大子序和

在状态转移时同时维护一下子序列的第一个和最后一个元素即可

# Code

```c++
class Solution {
public:
    int maxSubArray(vector<int>& nums) {
        int n=nums.size();

        vector<int>dp(n+1,0);
        int res=-0x3f3f3f3f;
        for(int i=0;i<n;i++)
        {
            if(dp[i]>0)dp[i+1]=nums[i]+dp[i];
            else dp[i+1]=nums[i];
            res=max(res,dp[i+1]);
        }
        return res;
    }
};
```
```c++
#include<bits/stdc++.h>
using namespace std;
int main(){
	int a[100];
	int n;
	cin>>n;
	for(int i=0;i<n;i++)cin>>a[i];
	int ans=a[0],num=a[0];//ans最终的和
	int low=0,anslow=0;
	int anshigh=0;
	for(int i=1;i<n;i++){
		if(num>0){
			num+=a[i];
		}
		else{
			num=a[i];
			low=i;
		}
		if(num>ans){
			ans=num;
			anslow=low;
			anshigh=i; 
		}
	}
	cout<<ans<<' '<<anslow<<' '<<anshigh<<endl; 
	return 0;
}
```
