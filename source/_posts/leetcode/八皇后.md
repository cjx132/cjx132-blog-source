# 综述

-   注意对角线和反对角线的表示方式（直线的截距）
-   另外，数组要开2倍的N,因为对角线的条数是2n-1

# Code

```c++
class Solution {
public:
    int col[20],dg[20],udg[20];
    vector<vector<string>>ans;
    vector<string>g;
    vector<vector<string>> solveNQueens(int n) {
        string t="";
        for(int i=0;i<n;i++)t+=".";
        for(int i=0;i<n;i++)g.push_back(t);
        dfs(0,n);
        return ans;
    }

    void dfs(int u,int n)
    {
        if(u==n)
        {
            ans.push_back(g);
            return;
        }
        for(int i=0;i<n;i++)
        {
            if(!col[i]&&!dg[u+i]&&!udg[u-i+n])
            {
                g[u][i]='Q';
                col[i]=1;dg[u+i]=1;udg[u-i+n]=1;
                dfs(u+1,n);
                col[i]=0;dg[u+i]=0;udg[u-i+n]=0;
                g[u][i]='.';
            }
        }
    }
};
```
