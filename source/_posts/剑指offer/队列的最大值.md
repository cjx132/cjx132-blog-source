# Code

```c++
class MaxQueue {
public:
    queue<int>q;
    int dq[10005];
    int hh=0,tt=-1;

    MaxQueue() {

    }
    
    int max_value() {
        if(hh<=tt)return dq[hh];
        else return -1;
    }
    
    void push_back(int value) {
        q.push(value);
        while(hh<=tt&&dq[tt]<=value)tt--;
        dq[++tt]=value;
    }
    
    int pop_front() {
        if(q.empty())return -1;
        int t=q.front();
        q.pop();
        if(hh<=tt&&dq[hh]==t)hh++;
        return t;
    }
};

/**
 * Your MaxQueue object will be instantiated and called as such:
 * MaxQueue* obj = new MaxQueue();
 * int param_1 = obj->max_value();
 * obj->push_back(value);
 * int param_3 = obj->pop_front();
 */
```
