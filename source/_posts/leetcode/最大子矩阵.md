# 综述

-   先回顾[连续子数组的最大和](https://cjx132.github.io/cjx132/2020/03/03/%E5%89%91%E6%8C%87offer/%E8%BF%9E%E7%BB%AD%E5%AD%90%E6%95%B0%E7%BB%84%E7%9A%84%E6%9C%80%E5%A4%A7%E5%92%8C/)
-   top,bottom双指针遍历所有可能的的两个“行对”，即子矩阵的上下两条边，这决定了矩阵的高
-   将top-bottom之间的每一列（求出每一列的累计的和）看成一维数组中的一项，在其中求最大子数组，即求出符合的子矩阵的宽

# Code

```c++
class Solution {
public:
    vector<int> getMaxMatrix(vector<vector<int>>& matrix) {
        vector<int>ans(4,0);
        int n=matrix.size(), m=matrix[0].size();
        vector<vector<int>>pre_sum(n+1,vector<int>(m,0));
        for(int i=1;i<=n;i++)
            for(int j=0;j<m;j++)
                pre_sum[i][j]=pre_sum[i-1][j]+matrix[i-1][j];
        
        int res=matrix[0][0];//初始化为左上角元素
        for(int top=0;top<n;top++)
            for(int bottom=top;bottom<n;bottom++)
            {
                vector<int>tmp;
                for(int k=0;k<m;k++)
                    tmp.push_back(pre_sum[bottom+1][k]-pre_sum[top][k]);
                
                int st=0,sum=tmp[0];
                for(int i=1;i<m;i++)
                {
                    if(sum>0)sum+=tmp[i];
                    else
                    {
                        sum=tmp[i];
                        st=i;
                    }
                    if(sum>res)
                    {
                        res=sum;
                        ans[0]=top;
                        ans[1]=st;
                        ans[2]=bottom;
                        ans[3]=i;
                    }
                }
            }
        return ans;
    }
};
```
