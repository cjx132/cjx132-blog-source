# Code

```plain
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    // int maxDepth(TreeNode* root) {//后序遍历返回最大值
    //     if(!root){
    //         return 0;
    //     }
    //     return max(maxDepth(root->left),maxDepth(root->right))+1;
    // }
    
     int maxDepth(TreeNode* root) {//层次遍历
        queue<TreeNode*>q;
        if(!root)return 0;
        q.push(root);
        int ans=0;
        while(!q.empty()){
            int n=q.size();//同一层次节点个数
            for(int i=0;i<n;i++){
                TreeNode* t=q.front();
                q.pop();
                if(t->left)q.push(t->left);
                if(t->right)q.push(t->right);
            }
            ans++;//每一层遍历完+1
        }
        return ans;
    }
};
```
