# 综述

-   快慢指针找到中点（靠后的一个）之后，反转后一半链表，之后拼接两个链表

# Code

```c++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* reverse(ListNode *head)
    {
        if(!head||!head->next)return head;

        auto res=reverse(head->next);
        head->next->next=head;
        head->next=nullptr;
        return res;
    }

    void reorderList(ListNode* head) {
        if(!head||!head->next)return;

        ListNode *a=head,*b=head;
        //找到中点
        while(a&&a->next)
        {
            a=a->next->next;
            b=b->next;
        }
        auto tb=b->next;
        b->next=nullptr;
        auto ta=head;
        //反转
        tb=reverse(tb);
        //合并
        ListNode* dummy=new ListNode(-1);
        ListNode *first=dummy;
        bool f=1;
        while(ta&&tb)
        {
            if(f)
            {
                first->next=ta;
                ta=ta->next;
            }
            else 
            {
                first->next=tb;
                tb=tb->next;
            }
            first=first->next;
            f=!f;
        }
        if(!ta)first->next=tb;
        if(!tb) first->next=ta;
        head=dummy->next;
    }
};
```
