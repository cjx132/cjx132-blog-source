```c++
class Solution {
public:
    int uniquePathsWithObstacles(vector<vector<int>>& obstacleGrid) {
        int ans=0;
        int n=obstacleGrid.size();
        int m=obstacleGrid[0].size();
        int f[105][105]={0};
        if(obstacleGrid[0][0]==0)f[0][0]=1;
        for(int i=1;i<n;i++)f[i][0]=obstacleGrid[i][0]==1?
            0:f[i-1][0];
        for(int j=1;j<m;j++)f[0][j]=obstacleGrid[0][j]==1?0:f[0][j-1];
        for (int i=1;i<n;i++)
            for (int j=1;j<m;j++)
                {
                    if(obstacleGrid[i][j]==1)
                        f[i][j]=0;
                    else f[i][j]=f[i-1][j]+f[i][j-1];
                }
        
        for (int i=0;i<n;i++)
        {
            for (int j=0;j<m;j++)
                cout<< f[i][j]<<' ';
            cout<<endl;
        }
        return f[n-1][m-1];
    }
};
```
