# Code

```c++
//宽度优先搜索
class Solution {
public:
    int numSquares(int n) {
        vector<int>dist(n+1,INT_MAX);
        queue<int>q;
        q.push(0);
        dist[0]=0;
        while(!q.empty()){
            int t=q.front();
            q.pop();
            if(t==n)return dist[n];
            for(int i=0;t+i*i<=n;i++){
                int j=t+i*i;
                if(dist[j]>dist[t]+1){
                    dist[j]=dist[t]+1;
                    q.push(j);
                }
            }
        }
        return 0;
    }
};
```
```c++
class Solution {
public:
    int f[10010];
    int numSquares(int n) {
        memset(f,0x3f,sizeof f);
        f[0]=0;
        for(int i=1;i<=n;i++)
            for(int j=1;i-j*j>=0;j++)
                f[i]=min(f[i],f[i-j*j]+1);
        return f[n];
    }
};
```

# 复杂度分析

时间复杂度：O(n\*sqrt(n))

空间复杂度：O(n)
