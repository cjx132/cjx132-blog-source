# 综述

-   [参考](https://www.acwing.com/solution/content/114/)
-   方法1类似于卡特兰数，需要倒着遍历一遍的原因是：因为val的值可能到底也清不到0 还大于0，而在这其中路过很多个合法的子括号串可能并不能统计出来，而反向来遍历就可以杜绝这种情况了
-   用栈模拟一遍，将所有匹配的括号的位置全部置1（类似于括号匹配），经过这样的处理后, 此题就变成了寻找最长的连续的1的长度

# Code

```c++
class Solution {
public:
    int longestValidParentheses(string s) {
        int n=s.size();
        int res=0,val=0,st=0;
        for(int i=0;i<n;i++)
        {                                                           
            if(s[i]=='(')val++;
            else val--;
            if(!val)res=max(res,i-st+1);
            else if(val<0)
            {
                st=i+1;
                val=0;
            }
        }
        st=n-1,val=0;
        for(int i=n-1;i>=0;i--)
        {
            cout<<s[i]<<endl;
            if(s[i]==')')val++;
            else val--;
            if(!val)res=max(res,st-i+1);
            else if(val<0)
            {
                st=i-1;
                val=0;
            }
        }
        return res;
    }
};
```
```c++
class Solution {
public:
    int longestValidParentheses(string s) {
        stack<int>st;
        int len=s.size();
        vector<int>mark(len,0);//记录配对索引
        int res=0;
        for (int i=0;i<s.size();i++){
            char c=s[i];
            if(c=='('){
                st.push(i);
            }else{
                if(st.empty())continue;//栈为空，说明此时不能匹配(此处为continue,与括号匹配有所不同)
                if(c==')'){
                    mark[i]=1;
                    mark[st.top()]=1;
                    st.pop();
                }
            }
        }
        //从标记数组中找出最大连续为1的长度(这种代码写法要牢记)
        int t=0;
        for(int i=0;i<len;i++){
            //以下if..else..块中代码写法要牢记
            if(mark[i]==1){//这个判断条件不同问题还可换成不同的判断逻辑
                t++;
                res=max(res,t);
            }else{
                t=0;
            }
        }
        return res;
    }
};
```
