# 综述

-   先做二叉树的直径
-   唯一的区别是：不是求边长，而是路径上节点的value之和，因为节点的value可能是负数，因此求左孩子右孩子的时候要`int leftSum = max(0, dfs(root.left));`**舍弃掉负数的结果。**

# Code

```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    int res=INT_MIN;
    int maxPathSum(TreeNode* root) {
        depth(root);
        return res;
    }
     int depth(TreeNode* root){
         //函数的作用是求得包含root节点的单边分支最大和
        if(!root)return 0;
        //计算左边分支最大值，左边分支如果为负数还不如不选择
        int left_depth=max(depth(root->left),0);
        //计算右边分支最大值，右边分支如果为负数还不如不选择
        int right_depth=max(depth(root->right),0);
        //全局找最大值
        res=max(res,left_depth+right_depth+root->val);
        //返回值需注意加上当前节点值
        return max(left_depth,right_depth)+root->val;
    }
};
```
