# Code

```c++
class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        vector<int>res;
        unordered_map<char,int>need;
        unordered_map<char,int>window;
        for(char c:p){
            need[c]++;
        }
        int left=0;
        int right=0;
        int flag=0;
        //区间左闭右开
        while(right<s.size()){
            //窗口右移
            char c=s[right];
            right++;
            //此时需要的更新操作
            if(need.count(c)){//这个判断不能少
            //以下两行代码顺序不能乱
                window[c]++;
                if(window[c]==need[c]){
                    flag++;
                }
            }
            
            while(right-left>=p.size()){//当窗口大小等于字符串p的大小时进入下面操作
                if(flag==need.size())res.push_back(left);//这里的判断条件是need的大小和flag相等，而不是p的大小
                //移动窗口
                char d=s[left];
                left++;
                if(need.count(d)){//这个判断不能少
                //以下代码顺序不能乱
                    if(window[d]==need[d]){
                        flag--;
                    }
                    window[d]--;
                }
            }
        }
        return res;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(n)
