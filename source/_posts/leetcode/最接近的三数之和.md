# Code

```c++
class Solution {
public:
    int threeSumClosest(vector<int>& nums, int target) {
        int res,mint=INT_MAX;
        sort(nums.begin(),nums.end());
        int n=nums.size();
        for(int k=0;k<n-2;k++){
            int i=k+1,j=n-1;
            while(i<j){
                int t=nums[k]+nums[i]+nums[j];
                if(abs(target-t)<mint){
                    res=t;
                    mint=abs(target-t);
                }else{
                    if(t>target)j--;
                    else i++;
                }
            }
        }
        return res;
    }
};
```

# 复杂度分析

时间复杂度：O(n^2)

空间复杂度：O(1)
