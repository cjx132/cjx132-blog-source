# 综述

-   观察规律可知，遍历方向由层数决定，而层数即为横纵坐标之和。
-   if,else不能颠倒(正方形情况)

# Code

```c++
class Solution {
public:
    vector<int> findDiagonalOrder(vector<vector<int>>& mat) {
        vector<int>ans;
        int row=mat.size();
        if (row==0)return ans;
        int col=mat[0].size();
        int r=0,c=0;
        for(int i=0;i<row*col;i++)
        {
            ans.push_back(mat[r][c]);
            //向上移动
            if((r+c)%2==0)
            {
                //前面两个if,else不能颠倒(正方形情况)
                if(c==col-1)r++;
                else if(r==0)c++;
                else
                {
                    c++;
                    r--;
                }
            }
            //向下移动
            else
            {
                //前面两个if,else不能颠倒(正方形情况)
                if(r==row-1)c++;
                else if(c==0)r++;
                else
                {
                    r++;
                    c--;
                }
            }
        }
        return ans;
    }
};
```
