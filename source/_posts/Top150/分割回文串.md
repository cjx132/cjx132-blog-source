# 综述

-   回顾一下求最长回文子串的操作
-   回溯的思路参考一下这个[题解](https://leetcode-cn.com/problems/palindrome-partitioning/solution/hui-su-you-hua-jia-liao-dong-tai-gui-hua-by-liweiw/)（有一个优化是预处理字符串，快速得出某个子串是否为回文串）
-   注意代码中的注释

# Code

```c++
class Solution {
public:
    vector<string>ss;
    vector<vector<string> >res;
    vector<vector<string>> partition(string s) {
        int n=s.size();
        vector<vector<int> >dp(n+1,vector<int>(n+1,0));
        for(int i=0;i<n;i++){
            //求最长回文子串模板
            help(i,i,s,dp);
            help(i,i+1,s,dp);
        }
        //回溯求解
        dfs(s,0,dp);
        return res;
    }

    void help(int left,int right,string& s,vector<vector<int> >&dp){
        //中心扩散
        while(left>=0&&right<s.size()&&s[left]==s[right]){
            dp[left][right]=1;
            left--;
            right++;
        }
    }

    void dfs(string &s,int left, vector<vector<int> >&dp){
        if(s.size()==left){
            res.push_back(ss);
            return;
        }
        for(int i=left;i<s.size();i++){
            string temp=s.substr(left,i-left+1);
            if(!dp[left][i]){//剪枝
                continue;//不能写为return,常见错误
            }
            ss.push_back(temp);
            dfs(s,i+1,dp);
            ss.pop_back();
        }
    }
};
```
