# Code

```c++
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* sortList(ListNode* head) {
        return merge_sort(head);
    }

    ListNode* merge_sort(ListNode* head)
    {
        if(!head||!head->next)return head;//空节点和单节点直接返回结果
        ListNode* first=head,*second=head;//快慢指针找到链表的中点
        while(first&&first->next&&first->next->next)
        {
            first=first->next->next;
            second=second->next;
        }
        auto a=second->next;
        second->next=nullptr;//必要
        auto b=head;
        auto sa=merge_sort(a);
        auto sb=merge_sort(b);
        return merge(sa,sb);
    }
    ListNode* merge(ListNode *a,ListNode *b)
    {//合并两个排序链表
        if(!a)return b;
        if(!b)return a;

        if(a->val<=b->val)
        {
            a->next=merge(a->next,b);
            return a;
        }
        else
        {
            b->next=merge(a,b->next);
            return b;
        }
    }
};
```

# 复杂度分析

时间复杂度：O(nlogn)

空间复杂度：O(1)
