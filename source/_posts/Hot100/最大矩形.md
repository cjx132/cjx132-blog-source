-   见柱状图中最大的矩形问题解析

# Code

```c++
class Solution {
public:
    int largestRectangleArea(vector<int> heights) {//这里传参本题不能使用引用，否则出错
        int n=heights.size();
        stack<int>stk;
        //左右边界值
        heights.insert(heights.begin(),-1);
        heights.push_back(-1);
        int res=0;
        for(int i=0;i<heights.size();i++)
        {
            while(!stk.empty()&&heights[i]<heights[stk.top()])
            {
                int t=stk.top();
                stk.pop();
                while(!stk.empty()&&heights[stk.top()]==heights[t])
                    stk.pop();
                if(!stk.empty())
                    res=max(res,(i-stk.top()-1)*heights[t]);
            }
            stk.push(i);
        }
        return res;
    }
    int maximalRectangle(vector<vector<char>>& matrix) {
        int m=matrix.size();
        if(!m)return 0;
        int n=matrix[0].size();
        int res=0;
        vector<int>heights(n,0);
        for(int i=0;i<m;i++)//逐行更新heights数组
        {
            for(int j=0;j<n;j++)
            {
                //以下if..else..块中代码写法要牢记
                if(matrix[i][j]=='1')//这个判断条件不同问题还可换成不同的判断逻辑
                    heights[j]++;
                else heights[j]=0;
            }
            res=max(res,largestRectangleArea(heights));
        }
        return res;
    }
};
```

# 复杂度分析

时间复杂度：O(m\*n)

空间复杂度：O(n)
