# Code

```c++
class Solution {
public:
    int translateNum(int num) {
        string str=to_string(num);
        int n=str.size();
        str=' '+str;

        int f[20];
        f[0]=1;;
        f[1]=1;
        for(int i=2;i<=n;i++)
        {
            if(str[i-1]=='1')f[i]=f[i-1]+f[i-2];
            else if(str[i-1]=='2'&&str[i]>='0'&&str[i]<='5')f[i]=f[i-1]+f[i-2];
            else f[i]=f[i-1];
        }
        return f[n];
    }
};
```
