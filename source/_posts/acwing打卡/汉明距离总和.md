# Code

```c++
class Solution {
public:
    int totalHammingDistance(vector<int>& nums) {
        int res=0;
        for(int i=0;i<32;i++){
            int t=0;
            for(auto num :nums){
                if((num>>i)&1)t++;
            }
            res+=t*(nums.size()-t);
        }
        return res;
    }
};
```
