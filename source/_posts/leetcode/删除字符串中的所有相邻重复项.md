# Code

```c++
class Solution {
public:
    string removeDuplicates(string S) {
        int n=S.size();
        int idx=-1;
        for(int i=0;i<n;i++)
        {
            if(idx==-1||S[idx]!=S[i])S[++idx]=S[i];
            else --idx;
        }
        return S.substr(0,idx+1);
    }
};
```
