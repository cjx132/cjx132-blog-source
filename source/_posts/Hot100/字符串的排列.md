# 综述

-   首先进行排序
    
-   注意如何剪枝
    
    > if(i>0&&s\[i\]==s\[i-1\]&&!flag\[i-1\])continue;
    > 
    > ```
    > //i>0保证i-1不越界，s[i]==s[i-1]保证第一个可以被递归
    > //flag[i-1]==0,代表i-1已经先被递归而且结束，然后回溯到开始的地方，此时后面如果再相等，则剪枝
    > ```

# Code

```c++
class Solution {
public:
    int flag[10]={0};
    vector<string>res;
    string t="";
    vector<string> permutation(string s) {
        sort(s.begin(),s.end());//先排序（字符串string排序方法）
        dfs(0,s,s.size());
        return res;
    }
    void dfs (int nowk,string s,int n){
        if(nowk==n){
            res.push_back(t);
            return;
        }
        for(int i=0;i<n;i++){
            if(flag[i]==1)continue;
            if(i>0&&s[i]==s[i-1]&&!flag[i-1])continue;
            //i>0保证i-1不越界，s[i]==s[i-1]保证第一个可以被递归
            //flag[i-1]==0,代表i-1已经先被递归，然后被回溯到这里，此时后面如果再相等，则跳过
            flag[i]=1;
            t+=s[i];
            dfs(nowk+1,s,n);
            t.pop_back();//字符串从尾部删除
            flag[i]=0;
        }
    }
};
```
