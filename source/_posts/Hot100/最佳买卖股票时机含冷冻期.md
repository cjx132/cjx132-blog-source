# Code

```c++
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int n=prices.size();
        vector<vector<int>>f(n,vector<int>(2,0));
        f[0][0]=0;
        f[0][1]=-prices[0];
        
        for(int i=1;i<n;i++)
        {
            f[i][0]=max(f[i-1][0],f[i-1][1]+prices[i]);
            //注意特判,f[-1][0]=0
            f[i][1]=max(f[i-1][1],(i>1?f[i-2][0]:0)-prices[i]);
        }
        return f[n-1][0];
    }
};
```
