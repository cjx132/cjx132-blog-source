# Code

```c++
class Solution {
public:
    int wiggleMaxLength(vector<int>& nums) {
        nums.erase(unique(nums.begin(),nums.end()),nums.end());
        if(nums.size()<=2)return nums.size();
        int res=2;
        for(int i=1;i+1<nums.size();i++){
            int l=nums[i-1],mid=nums[i],r=nums[i+1];
            if(l<mid&&mid>r)res++;
            if(l>mid&&mid<r)res++;
        }
        return res;
    }
};
```
