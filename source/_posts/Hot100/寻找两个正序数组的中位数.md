# Code

```c++
class Solution {
public:
    double findMedianSortedArrays(vector<int>& nums1, vector<int>& nums2) {
        int n=nums1.size()+nums2.size();
        if(n&1)return find_k(nums1,0,nums2,0,n/2+1);
        else return (find_k(nums1,0,nums2,0,n/2)+find_k(nums1,0,nums2,0,n/2+1))*0.5;
    }
    /*
    函数作用:寻找nums1和nums2两个数组总的第k小的数字
    下标[st1,num1.size()]是nums1查找的范围,下标[st2,nums2.size()]是nums2查找的范围
    */
    double find_k(vector<int>& nums1,int st1,vector<int>& nums2,int st2,int k)
    {
        //保证nums1的长度小于nums2的长度
        int len1=nums1.size()-st1,len2=nums2.size()-st2;
        if(len1>len2)return find_k(nums2,st2,nums1,st1,k);

        if(len1==0)return nums2[st2+k-1];//边界情况,nums1长度为0,直接在nums2数组中查找
        if(k==1)return min(nums1[st1],nums2[st2]);//k=1直接找到开头的最小值就行
        
        int i=st1+min(k/2,len1)-1;//nums1数组中第k/2小的元素的下标,有可能超出范围,要取一个min
        int j=st2+k/2-1;//nums2数组中第k/2小的元素的下标

        if(nums1[i]<nums2[j])//此时[st1,i]区间的元素一定在第k小之内
        {
            k=k-(i-st1+1);
            return find_k(nums1,i+1,nums2,st2,k);
        }
        else//此时[j,st2]区间的元素一定在第k小之内
        {
            k=k-(j-st2+1);
            return find_k(nums1,st1,nums2,j+1,k);
        }            
    }
};
```

# 复杂度分析

时间复杂度：O(log(m+n))

空间复杂度：O(1)
