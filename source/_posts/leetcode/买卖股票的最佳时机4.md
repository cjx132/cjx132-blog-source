# Code

```c++
class Solution {
public:
    int maxProfit(int k, vector<int>& prices) {
        int len=prices.size();
        if(len==0||k==0)return 0;

        if(k>len/2){
            return maxProfit1(prices);
        }

        // vector<vector<vector<int> > >dp(len,vector<vector<int> >(k+1,vector<int> (2,0)));

        // for(int i=1;i<=k;i++){
        //     dp[0][i][1]=-prices[0];
        //     dp[0][i][0]=0;
        // }
        
        // for(int i=1;i<len;i++){
        //     for(int j=1;j<=k;j++){
        //         dp[i][j][0]=max(dp[i-1][j][0],dp[i-1][j][1]+prices[i]);
        //         dp[i][j][1]=max(dp[i-1][j][1],dp[i-1][j-1][0]-prices[i]);
        //     }
        // }
        // return dp[len-1][k][0];
        vector<vector<int> >dp(k+1,vector<int>(2,0));
        
        for(int i=1;i<=k;i++){
            dp[i][0]=0;
            dp[i][1]=-prices[0];
        }

        for(int i=1;i<len;i++){
            for(int j=1;j<=k;j++){
                dp[j][0]=max(dp[j][0],dp[j][1]+prices[i]);
                dp[j][1]=max(dp[j][1],dp[j-1][0]-prices[i]);
            }
        }
        return dp[k][0];
    }

    int maxProfit1(vector<int>& prices) {//k不限次数的情况
        int len=prices.size();
        if(len==0)return 0;

        int un_hold=0;
        int hold=-prices[0];

        for(int i=1;i<len;i++){
            un_hold=max(un_hold,hold+prices[i]);
            hold=max(hold,un_hold-prices[i]);   
        }

        return un_hold;
    }
};
```

# 复杂度分析

-   时间复杂度：O(nk)
    
-   空间复杂度：O(k)
