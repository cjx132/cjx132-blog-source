# Code

```c++
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        int m=matrix.size(),n=matrix[0].size();

        int dx[4]={-1,0,1,0},dy[4]={0,1,0,-1};

        int d=1;

        int x=0,y=0;
        vector<int>ans;
        for(int i=1;i<=m*n;i++)
        {
            ans.push_back(matrix[x][y]);
            matrix[x][y]=0x3f3f3f3f;
            int tx=dx[d]+x,ty=dy[d]+y;
            if(tx<0||tx>=m||ty<0||ty>=n||matrix[tx][ty]==0x3f3f3f3f)
            {
                d=(d+1)%4;
                tx=dx[d]+x,ty=dy[d]+y;
            }
            x=tx,y=ty;
        }
        return ans;
    }
};
```
