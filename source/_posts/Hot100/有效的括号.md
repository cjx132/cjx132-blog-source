# Code

```c++
class Solution {
public:
    char st[10010];
    int tt=0;
    bool isValid(string s) {
        int n=s.size();
        for(int i=0;i<n;i++)
        {
            if(s[i]=='['||s[i]=='{'||s[i]=='(')st[++tt]=s[i];
            else 
            {
                if(tt<=0)return false;
                if((st[tt]=='('&&s[i]==')')||(st[tt]=='['&&s[i]==']')||(st[tt]=='{'&&s[i]=='}'))--tt;
                else return false;
            }
        }
        return tt<=0;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(n)
