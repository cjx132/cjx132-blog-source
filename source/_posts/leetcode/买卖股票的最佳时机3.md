```c++
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int n=prices.size();
        vector<vector<vector<int>>>f(n,vector<vector<int>>(3,vector<int>(2,0)));
        for(int k=1;k<=2;k++)
        {
            f[0][k][0]=0;
            f[0][k][1]=-prices[0];
        }

        for(int i=1;i<n;i++)
            for(int k=1;k<=2;k++)
            {
                f[i][k][0]=max(f[i-1][k][0],f[i-1][k][1]+prices[i]);
                f[i][k][1]=max(f[i-1][k][1],f[i-1][k-1][0]-prices[i]);
            }
        return f[n-1][2][0];
    }
};
```
```c++
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int len=prices.size();
        if(len==0)return 0;
        int dp20=0;
        int dp21=-prices[0];
        int dp10=0;
        int dp11=-prices[0];
        
        for(int i=1;i<len;i++){
            dp20=max(dp20,dp21+prices[i]);
            dp21=max(dp21,dp10-prices[i]);
            dp10=max(dp10,dp11+prices[i]);
            dp11=max(dp11,-prices[i]);
        }
        return dp20;
    }
};
```

# 复杂度分析

-   时间复杂度：O(n)
    
-   空间复杂度：O(1)
