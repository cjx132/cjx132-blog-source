# 综述

-   **带权重的并查集（注意权重的路径压缩，以及两个集合合并时权重的更新）**
    
-   四边形定则
    
    ![](https://pic.leetcode-cn.com/be1550bb034405199acad6201faeb2be1d6d5a9f80748ef3080a1d01fe028ffd-%E5%9B%BE%E7%89%87.png)
    

# Code

```c++
class Solution {
public:
    int farther[27];
    double weight[27];
    vector<double> calcEquation(vector<vector<string>>& equations, vector<double>& values, vector<vector<string>>& queries) {
        vector<double>res;
        unordered_map<string,int>mp;//string和数字的映射，并查集使用传统的数字进行操作
        int num=0;//用数字代表不同的字母
        for(int i=1;i<=26;i++){//并查集初始化
            farther[i]=i;
            weight[i]=1.0;
        }
        for(int i=0;i<equations.size();i++){
            if(mp.find(equations[i][0])==mp.end()){
                mp[equations[i][0]]=++num;
            }
            if(mp.find(equations[i][1])==mp.end()){
                mp[equations[i][1]]=++num;
            }
            //形成集合
            Union(mp[equations[i][0]],mp[equations[i][1]],values[i]);
        }
        for(int i=0;i<queries.size();i++){
            if(mp.find(queries[i][0])==mp.end()||mp.find(queries[i][1])==mp.end()){
                //出现未出现过的字母
                res.push_back(-1);
                continue;
            }
            int d1=mp[queries[i][0]];
            int d2=mp[queries[i][1]];
            int root1=findroot(d1);
            int root2=findroot(d2);
            if(root1!=root2){
                //不属于同一集合，不能求解
                res.push_back(-1);
                continue;
            }
            //a/b = (a/x1)/(b/x1),根节点相同时计算相应结果
            res.push_back(weight[d1]/weight[d2]);
        }
        return res;
    }
    //并
    void Union(int x,int y,double v){
        int rootx=findroot(x);
        int rooty=findroot(y);
        if(rootx!=rooty){//判断是否属于一个集合（否则将一个集合的元素进行合并就形成了环）
            farther[rootx]=rooty;//注意这里是把rootx和rooty相连，并查集容易出错点（否则将无法合并这两个不同的集合）
            weight[rootx]=v*weight[y]/weight[x];//四边形定则，详细解释在图中
        }
    }
    //查
    int findroot(int x){
        if(x==farther[x]){
            return x;
        }
        int t=findroot(farther[x]);
        //路径压缩
        weight[x]*=weight[farther[x]];
        farther[x]=t;
        return t;
    }
};
```

# 复杂度分析

时间复杂度：O(n)

空间复杂度：O(1)
