# Code

```plain
class Solution {
public:
    int left_bound(vector<int>& nums, int target){
        int high=nums.size();//注意点1
        int low=0;
        int mid;
        while(low<high){//注意点2，左闭右开
            mid=low+(high-low) /2;//注意点3：防止溢出
            if(nums[mid]>target)high=mid;//注意点:4，左闭右开
            else if(nums[mid]<target)low=mid+1;//注意点5，除法向下取整，一般保证low=mid+1
            else if(nums[mid]==target)high=mid;//注意点6
        }
        if(high==nums.size())return -1;//注意点7，high和low此时相等，取值范围是[0,nums.size()]
        return (nums[high]==target)?high:-1;
    }
    int right_bound(vector<int>& nums, int target){
        int high=nums.size();
        int low=0;
        int mid;
        while(low<high){
            mid=low+(high-low) /2;
            if(nums[mid]<=target)low=mid+1;
            else high=mid;
        }
        if(high==0)return -1;
        return (nums[high-1]==target)?high-1:-1;//注意点8，求右边界需要-1
    }
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int>ans;
        ans.push_back(left_bound(nums,target));
        ans.push_back(right_bound(nums,target));
        return ans;
    }
};
```
