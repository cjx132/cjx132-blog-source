# Code

中心扩散法

```c++
class Solution {
public:
    string help(string &s, int l,int r){
        int n=s.size();
        while(l>=0&&r<n&&s[l]==s[r]){//注意边界条件的处理
            l--;
            r++;
        }
        return s.substr(l+1,r-l-1);
    }
    string longestPalindrome(string s) {
        int n=s.size();
        string res;
        for(int i=0;i<n;i++){
            string s1=help(s,i,i);
            string s2=help(s,i,i+1);
            res=res.size()>s1.size()? res :s1;
            res=res.size()>s2.size()? res :s2;
        }
        return res;
    }
};
```

字符串哈希

```c++
class Solution {
public:
    typedef unsigned  long long ULL;
    static const int N=2010;//开为2倍
    ULL p[N],hl[N],hr[N];
    int base=131;
    char str[N];

    ULL get(ULL h[],int l,int r)
    {
        return h[r]-h[l-1]*p[r-l+1];
    }

    string longestPalindrome(string s) {
        int n=s.size();
        for(int i=2*n;i>0;i-=2)
        {
            str[i]=s[i/2-1];
            str[i-1]='#';
        }//从1开始
        // puts(str+1);

        n*=2;
        p[0]=1;//不能少
        for(int i=1,j=n;i<=n;i++,j--)
        {
            p[i]=p[i-1]*base;
            hl[i]=hl[i-1]*base+str[i]-'a'+1;
            hr[i]=hr[i-1]*base+str[j]-'a'+1;
        }

        int res=0,al,ar;
        for(int i=1;i<=n;i++)
        {
            int l=0,r=min(i-1,n-i);
            while(l<r)
            {
                int mid=(l+r+1)/2;
                if(get(hl,i-mid,i-1)!=get(hr,n-(i+mid)+1,n-(i+1)+1))r=mid-1;
                else l=mid;
            }

            int cur_len=(str[i-l]=='#'?l:l+1);
            if(cur_len>res)
            {
                al=i-l,ar=i+l;
                res=cur_len;
            }
        }
        string ans;
        for(int i=al;i<=ar;i++)
            if(str[i]!='#')
                ans+=str[i];
        return ans;
    }
};
```

# 复杂度分析

时间复杂度：O(n^2)

空间复杂度：O(1)
