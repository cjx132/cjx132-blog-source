# 综述

先把所有字符串组合成一个完整的字符串，之间用\\n隔开，然后遍历字符串，遇到//时，删除到\\n, 遇到`/*`，删除到`*/`，最后的字符串再用\\n分割开。就是结果。

# Code

```c++
class Solution {
public:
    string get_str(string &s, int st ,int ed)
    {
        string res="";
        for(int i=st;i<=ed;i++)
            res+=s[i];
        return res;
    }

    vector<string> removeComments(vector<string>& source) {
        vector<string>ans;
        string all;
        for (auto &s : source)
        {
            s+="\n";
            all+=s;
        }
        int i=0;
        while(i<all.size()-1)
        {
            char fi=all[i],se=all[i+1];
            if(fi=='/'&&se=='/')
            {
                int j=i;
                while(j<all.size()&&all[j]!='\n')j++;
                all.erase(i,j-i);//保留"/n"
                i--;
            }
            else if(fi=='/'&&se=='*')
            {
                int j=i+2;
                while(j<all.size()-1&&!(all[j]=='*'&&all[j+1]=='/'))j++;
                all.erase(i,j+2-i);
                i--;
            }
            i++;
        }
        
        //双指针模板实现split
        int len=all.size();
        for(int i=0;i<len;i++)
        {
            int j=i;
            while(j<len&&all[j]!='\n')j++;
            string t=get_str(all,i,j-1);
            if(t!="")
                ans.push_back(t);
            i=j;
        }
        return ans;
    }
};
```

# 复杂度分析

-   时间复杂度：O(n)
    
-   空间复杂度：O(1)
