# Code

方法1：

![题解图](https://cjx132.github.io/picture-bed/img/image-20201121203537206.png)

-   绿色线条为第一轮替换
-   红色线条为第二轮替换

```c++
class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        int n=nums.size();

        int count=0;//记录交换次数（交换n次之后就完成了）
        int start_index=0;//记录每次从哪个位置开始一轮交换

        while(count<n){//结束条件要注意
            
            int cur_index=start_index;//记录当前轮到哪个位置去交换
            int cur_value=nums[cur_index];//记录cur_index的值
            
            do{
                int next_index=(cur_index+k)%n;//记录当前元素要交换到哪里
                int next_value=nums[next_index];//提前保存一下next_index位置的值
                
                nums[next_index]=cur_value;//交换成功
                count++;//更新交换次数

                //更新当前要交换的下标和值
                cur_index=next_index;
                cur_value=next_value;
                
            }while(start_index!=cur_index);//结束条件要注意
            start_index++;//注意更新
        }
    }
};
```

方法2：

```c++
class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        int n=nums.size();

        k%=n;
        reverse(nums.begin(),nums.end());
        reverse(nums.begin(),nums.begin()+k);
        reverse(nums.begin()+k,nums.end());
    }
};
```

# 复杂度分析

-   时间复杂度：O(n)
-   空间复杂度：O(1)
