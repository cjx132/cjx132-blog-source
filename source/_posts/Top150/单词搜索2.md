# 综述

-   结合不紧密的Trie+回溯（超时）
    
    > 1.  前缀树中search的修改：目的是去除最终结果中的重复元素
    > 2.  利用startWith进行剪枝
    > 3.  找到匹配的单词后不能`return`，因为可能有以当前单词为前缀的其他单词，如果`return`就不能找到它们了
    
-   结合紧密的Trie+回溯
    
    -   判断单词是否匹配？ 看node->word!=””
    -   判断当前字符串是否是某一单词的前缀？ 看node->next\[c-‘a’\]==nullptr
    -   下一个位置的字符串是？ node=node->next\[c-‘a’\]
-   [参考题解](https://leetcode-cn.com/problems/word-search-ii/solution/san-chong-jie-fa-you-qian-dao-shen-bao-sou-triebao/)
    

# Code

### 结合不紧密的Trie+回溯（超时）

```c++
class Trie{
private:
    bool isEnd;
    Trie* next[26];
public:
    Trie(){
        isEnd=false;
        memset(next,0,sizeof(next));
    }

    void insert(string str){
        Trie* node =this;
        for(auto c:str){
            if(node->next[c-'a']==NULL){
                node->next[c-'a']=new Trie();
            }
            node=node->next[c-'a'];
        }
        node->isEnd=true;
    }

    bool search(string str){
        Trie* node=this;
        for(auto c:str){
            if(node->next[c-'a']==NULL){
                return false;
            }
            node=node->next[c-'a'];
        }

        //这里修改了一下原来搜索的代码，如果找到之后，在返回true之前，置为false，保证最终的结果不会重复
        if(node->isEnd){
            node->isEnd=false;
            return true;
        }

        return false;
    }

    bool startWith(string str){
        Trie* node=this;
        for(auto c:str){
            if(node->next[c-'a']==NULL){
                return false;
            }
            node=node->next[c-'a'];
        }
        return true;
    }
};

class Solution {
public:
    int step[2][4]={{0,1,0,-1},{1,0,-1,0}};
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        vector<string> res;
        Trie* node=new Trie();
        for(auto str:words){
            node->insert(str);
        }
        string str="";
        for(int i=0;i<board.size();i++){
            for(int j=0;j<board[0].size();j++){
                dfs(board,node,i,j,res,str);
            }
        }
        return res;
    }
    void dfs(vector<vector<char>>& board, Trie*& node, int x, int y, vector<string>& res, string& str){
        if(x<0||y<0||x>=board.size()||y>=board[0].size()||board[x][y]=='.')return;
        if(!node->startWith(str))return;//利用前缀剪枝
        char c=board[x][y];
        str.push_back(c);
        if(node->search(str)){
            res.push_back(str);
        }
        board[x][y]='.';
        for(int i=0;i<4;i++){
            int tx=x+step[0][i];
            int ty=y+step[1][i];
            dfs(board,node,tx,ty,res,str);
        }
        board[x][y]=c;
        str.pop_back();
    }
};
```

### 结合紧密的Trie+回溯

```c++
class Trie{
public:
    string word;//前缀树的变形
    Trie* next[26];
    Trie(){
        word="";
        for(int i=0;i<26;i++){
            next[i]=nullptr;
        }
    }

    void insert(string str){
        Trie* node =this;
        for(auto c:str){
            if(node->next[c-'a']==NULL){
                node->next[c-'a']=new Trie();
            }
            node=node->next[c-'a'];
        }
        node->word=str;
    }
};

class Solution {
public:
    int step[2][4]={{0,1,0,-1},{1,0,-1,0}};
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        vector<string> res;
        Trie* node=new Trie();
        for(auto str:words){
            node->insert(str);
        }
        for(int i=0;i<board.size();i++){
            for(int j=0;j<board[0].size();j++){
                dfs(board,node,i,j,res);
            }
        }
        return res;
    }
    void dfs(vector<vector<char>>& board, Trie* node, int x, int y, vector<string>& res){
        if(x<0||y<0||x>=board.size()||y>=board[0].size()||board[x][y]=='.')return;
        char c=board[x][y];
        node=node->next[c-'a'];//node不是引用变量，也不是全局变量，改变之后不需要回溯
        if(node==nullptr)return;//这里相当于前缀剪枝
        
        if(node->word!=""){//这里相当于查找到
            res.push_back(node->word);
            node->word="";//这里实现最终结果的去重
            //这里不能有return,不是结束条件，继续搜索仍有可能有满足条件的结果
        }

        board[x][y]='.';
        for(int i=0;i<4;i++){
            int tx=x+step[0][i];
            int ty=y+step[1][i];
            dfs(board,node,tx,ty,res);
        }
        board[x][y]=c;
    }
};
```
