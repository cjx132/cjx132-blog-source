# Code

```c++
class Solution {
public:
    string multiply(string num1, string num2) {
        int na = num1.size(), nb = num2.size();
        reverse(num1.begin(), num1.end()), reverse(num2.begin(), num2.end());
        vector<int> C(na + nb,0);
        for (int i = 0; i < na; i++)
            for (int j = 0; j < nb; j++)
            {
                int p1 = i + j, p2 = i + j + 1;
                int t = (num1[i] - '0') * (num2[j] - '0') + C[p1];
                C[p1] = t % 10;
                C[p2] += t / 10;
            }

        while(C.size()>1 && C.back() == 0)C.pop_back();
        string res;
        for (int i = C.size() - 1; i >= 0 ; i--)res += to_string(C[i]);
        return res;
    }
};
```
