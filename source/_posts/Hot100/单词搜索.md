# 综述

错误示例

```python
# 这一段代码是错误的，不要模仿
for i in range(m):
    for j in range(n):
        # 对每一个格子都从头开始搜索
        return self.__search_word(board, word, 0, i, j, marked, m, n)
```

正确写法

```python
for i in range(m):
    for j in range(n):
        # 对每一个格子都从头开始搜索
        if self.__search_word(board, word, 0, i, j, marked, m, n):
            return True
```

以上写法是**搜到答案就返回**的写法，同时保证**每种可能性都搜到**

# Code

```c++
class Solution {
public:
    int dx[4]={-1,0,1,0},dy[4]={0,1,0,-1};
    bool exist(vector<vector<char>>& board, string word) {
        int m=board.size(),n=board[0].size();
        for(int i=0;i<m;i++)
            for(int j=0;j<n;j++)
                if(dfs(board,word,i,j,0))
                    return true;
        return false;
    }
    bool dfs(vector<vector<char>>& board,string &word,int x,int y,int u)
    {
        //两个判断的顺序不能颠倒
        if(word[u]!=board[x][y])return false;
        if(u==word.size()-1)return true;
        
        char c=board[x][y];
        board[x][y]='#';
        for(int i=0;i<4;i++)
        {
            int tx=x+dx[i],ty=y+dy[i];
            if(tx>=0&&tx<board.size()&&ty>=0&&ty<board[0].size())
                if(dfs(board,word,tx,ty,u+1))
                    return true;
        }
        board[x][y]=c;
        return false;
    }
};
```
